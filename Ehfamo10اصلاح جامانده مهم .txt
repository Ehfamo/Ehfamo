class Correlation:

def setup_logging() -> logging.Logger:
    logger = logging.getLogger(SYSTEM_NAME)
    logger.setLevel(logging.INFO)

    handler = logging.StreamHandler()
    fmt = logging.Formatter(
        fmt="%(asctime)s %(levelname)s %(name)s %(message)s "
            "trace_id=%(trace_id)s op=%(op)s"
    )
    handler.setFormatter(fmt)

    class ContextFilter(logging.Filter):
        def filter(self, record: logging.LogRecord) -> bool:
            if not hasattr(record, "trace_id"):
                record.trace_id = "no-trace"
            if not hasattr(record, "op"):
                record.op = "none"
            return True

    handler.addFilter(ContextFilter())
    logger.handlers = [handler]
    return logger

LOGGER = setup_logging()

@dataclass(frozen=True)
class TxSettings:
    statement_timeout_ms: int = DB_STATEMENT_TIMEOUT_MS
    lock_timeout_ms: int = DB_LOCK_TIMEOUT_MS
    deadlock_max_retries: int = DEADLOCK_MAX_RETRIES
    base_backoff_sec: float = DEADLOCK_BASE_BACKOFF_SEC
    max_backoff_sec: float = DEADLOCK_MAX_BACKOFF_SEC

def _sqlstate(e: Exception) -> Optional[str]:
    # psycopg exceptions usually have .sqlstate
    return getattr(e, "sqlstate", None)

def _is_retryable_pg_error(e: Exception) -> bool:
    # Deadlock detected: 40P01
    # Serialization failure: 40001
    # Lock not available: 55P03 (optional)
    s = _sqlstate(e)
    return s in {"40P01", "40001", "55P03"}

class DB:
    def __init__(self, dsn: str):
        self.pool = ConnectionPool(conninfo=dsn, min_size=1, max_size=10, kwargs={"row_factory": dict_row})

    def close(self) -> None:
        self.pool.close()

    def with_conn(self) -> Connection:
        return self.pool.getconn()

    def put_conn(self, conn: Connection) -> None:
        self.pool.putconn(conn)

    def run_tx(self, op: str, fn: Callable[[Connection], Any], *, settings: TxSettings, trace_id: str) -> Any:
        for attempt in range(settings.deadlock_max_retries + 1):
            conn: Optional[Connection] = None
            try:
                conn = self.with_conn()
                with conn.transaction():
                    # Timeouts are LOCAL to transaction
                    with conn.cursor() as cur:
                        cur.execute("SET LOCAL statement_timeout = %s", (settings.statement_timeout_ms,))
                        cur.execute("SET LOCAL lock_timeout = %s", (settings.lock_timeout_ms,))
                result = fn(conn)
                return result
            except Exception as e:
                # Never swallow BaseException (KeyboardInterrupt/SystemExit)
                if isinstance(e, BaseException) and not isinstance(e, Exception):
                    raise

                if _is_retryable_pg_error(e) and attempt < settings.deadlock_max_retries:
                    DB_RETRY.labels(reason=_sqlstate(e) or "unknown").inc()
                    backoff = min(settings.max_backoff_sec, settings.base_backoff_sec * (2 ** attempt))
                    jitter = backoff * (0.5 + random.random())
                    LOGGER.warning(
                        "db_retry",
                        extra={"trace_id": trace_id, "op": op},
                    )
                    time.sleep(jitter)
                    continue

                LOGGER.exception("db_tx_failed", extra={"trace_id": trace_id, "op": op})
                raise
            finally:
                if conn is not None:
                    self.put_conn(conn)

# SECURITY: KEYRING + AES-GCM + ROTATION

@dataclass(frozen=True)
class KeyMaterial:
    key_id: str
    key_bytes: bytes  # 32 bytes for AES-256-GCM

class Keyring:
    """
    Supports rotation by keeping multiple keys in environment.
    - Active key id: env ACTIVE_KEY_ENV
    - Key materials: env CRED_KEY_<key_id> = base64(32 bytes)
    """
    def __init__(self):
        active = os.environ.get(ACTIVE_KEY_ENV, "").strip()
        if not active or not KEY_ID_RE.match(active):
            raise RuntimeError(f"Missing/invalid active key id env: {ACTIVE_KEY_ENV}")
        self.active_key_id = active
        self.keys: Dict[str, bytes] = {}
        for k, v in os.environ.items():
            if not k.startswith(KEYRING_ENV_PREFIX):
                continue
            key_id = k[len(KEYRING_ENV_PREFIX):]
            if not KEY_ID_RE.match(key_id):
                continue
            raw = base64.b64decode(v)
            if len(raw) != 32:
                raise RuntimeError(f"Key {key_id} must be 32 bytes (decoded)")
            self.keys[key_id] = raw
        if self.active_key_id not in self.keys:
            raise RuntimeError(f"Active key {self.active_key_id} not found in keyring envs")

    def encrypt(self, plaintext: bytes, aad: bytes) -> bytes:
        key = self.keys[self.active_key_id]
        nonce = os.urandom(12)
        ct = AESGCM(key).encrypt(nonce, plaintext, aad)
        # envelope: key_id_len|key_id|nonce|ct
        kid = self.active_key_id.encode()
        return bytes([len(kid)]) + kid + nonce + ct

    def decrypt(self, blob: bytes, aad: bytes) -> bytes:
        if len(blob) < 1:
            raise ValueError("invalid_blob")
        kid_len = blob[0]
        if len(blob) < 1 + kid_len + 12 + 16:
            raise ValueError("invalid_blob_len")
        kid = blob[1:1+kid_len].decode()
        nonce = blob[1+kid_len:1+kid_len+12]
        ct = blob[1+kid_len+12:]
        key = self.keys.get(kid)
        if not key:
            raise ValueError("unknown_key_id")
        return AESGCM(key).decrypt(nonce, ct, aad)

# ============================================================

class RateLimiter:
    """
    DB-backed token bucket to work across multiple orchestrator instances.
    Schema: rate_limits(key text primary key, tokens double precision, last_update timestamptz, burst int)
    """
    def __init__(self, db: DB):
        self.db = db

    def allow(self, key: str, *, tps: float, burst: int, trace_id: str) -> bool:
        now = time.time()

        def _op(conn: Connection) -> bool:
            with conn.cursor() as cur:
                # Upsert with refill
                cur.execute("""
                    INSERT INTO rate_limits(key, tokens, last_update, burst)
                    VALUES (%s, %s, NOW(), %s)
                    ON CONFLICT (key) DO UPDATE SET
                        tokens = LEAST(
                            rate_limits.burst,
                            rate_limits.tokens +
                            EXTRACT(EPOCH FROM (NOW() - rate_limits.last_update)) * %s
                        ),
                        last_update = NOW(),
                        burst = EXCLUDED.burst
                    RETURNING tokens
                """, (key, float(burst), int(burst), float(tps)))
                row = cur.fetchone()
                tokens = float(row["tokens"])

                if tokens < 1.0:
                    return False

                cur.execute("""
                    UPDATE rate_limits
                    SET tokens = tokens - 1
                    WHERE key = %s
                """, (key,))
                return True

        ok = self.db.run_tx("rate_limit", _op, settings=TxSettings(), trace_id=trace_id)
        if not ok:
            RATE_LIMIT_BLOCK.labels(key=key).inc()
        return ok

# TASK TRANSITION (FOR UPDATE + ATOMIC AUDIT INSERT VIA CTE)
# ============================================================

class TaskTransitions:
    def __init__(self, db: DB):
        self.db = db

    def transition(self, task_id: str, to_stage: Stage, trace_id: str) -> None:
        def _op(conn: Connection) -> None:
            with conn.cursor() as cur:
                # Lock the task row to prevent race conditions
                cur.execute("""
                    SELECT task_id, current_stage, status
                    FROM task_queue
                    WHERE task_id = %s
                    FOR UPDATE
                """, (task_id,))
                row = cur.fetchone()
                if not row:
                    raise TaskNotFoundError(task_id)

                # Atomic update + audit log in ONE statement (no crash window)
                cur.execute("""
                    WITH updated AS (
                        UPDATE task_queue
                        SET current_stage = %s,
                            updated_at = NOW()
                        WHERE task_id = %s
                        RETURNING task_id, current_stage
                    )
                    INSERT INTO audit_log (audit_id, task_id, event_type, payload, created_at)
                    SELECT gen_random_uuid()::text,
                           task_id,
                           'STAGE_TRANSITION',
                           jsonb_build_object('to', current_stage),
                           NOW()
                    FROM updated
                """, (to_stage.value, task_id))

        self.db.run_tx("task_transition", _op, settings=TxSettings(), trace_id=trace_id)

@dataclass
class ConversationContext:
    conversation_id: str
    messages: List[Dict[str, str]] = field(default_factory=list)

class ConversationStore:
    def __init__(self, db: DB):
        self.db = db

    def add_message(self, ctx: ConversationContext, msg: Dict[str, str], trace_id: str) -> None:
        ctx.messages.append(msg)

        if len(ctx.messages) <= ARCHIVE_THRESHOLD_MESSAGES:
            return

        to_archive = ctx.messages[:-MAX_CONTEXT_MESSAGES]
        keep = ctx.messages[-MAX_CONTEXT_MESSAGES:]

        def _op(conn: Connection) -> None:
            with conn.cursor() as cur:
                # Archive is best-effort but must not lose data:
                # if archive fails, we do NOT truncate. (Fixes "archive failure loses history")
                cur.execute("""
                    INSERT INTO conversation_archive (archive_id, conversation_id, payload, created_at)
                    VALUES (%s, %s, %s::jsonb, NOW())
                """, (f"arch_{uuid.uuid4().hex}", ctx.conversation_id, json.dumps(to_archive)))

        try:
            self.db.run_tx("conversation_archive", _op, settings=TxSettings(), trace_id=trace_id)
            ctx.messages = keep
        except Exception:
            # Do not truncate if archive failed; preserve correctness.
            LOGGER.exception("archive_failed_no_truncate", extra={"trace_id": trace_id, "op": "conversation_archive"})

# ADAPTER REGISTRY (TTL EVICTION + THREAD-SAFE)
# ============================================================

@dataclass
class AdapterEntry:
    adapter_id: str
    adapter_obj: Any
    expires_at: float

class AdapterRegistry:
    def __init__(self):
        self._lock = threading.Lock()
        self._adapters: Dict[str, AdapterEntry] = {}

    def register(self, adapter_id: str, adapter_obj: Any) -> None:
        with self._lock:
            self._adapters[adapter_id] = AdapterEntry(
                adapter_id=adapter_id,
                adapter_obj=adapter_obj,
                expires_at=time.time() + ADAPTER_TTL_SEC,
            )

    def get(self, adapter_id: str) -> Optional[Any]:
        with self._lock:
            e = self._adapters.get(adapter_id)
            if not e:
                return None
            if time.time() > e.expires_at:
                self._adapters.pop(adapter_id, None)
                return None
            return e.adapter_obj

    def cleanup(self) -> int:
        now = time.time()
        removed: List[AdapterEntry] = []
        with self._lock:
            for k, e in list(self._adapters.items()):
                if now > e.expires_at:
                    removed.append(e)
                    self._adapters.pop(k, None)

        # Best-effort disconnect outside lock
        for e in removed:
            try:
                if hasattr(e.adapter_obj, "disconnect"):
                    e.adapter_obj.disconnect()
            except Exception:
                LOGGER.exception("adapter_disconnect_failed", extra={"trace_id": "no-trace", "op": "adapter_cleanup"})
        return len(removed)

class CircuitBreakerStore:
    def __init__(self, db: DB):
        self.db = db

    def load(self, name: str, trace_id: str) -> Dict[str, Any]:
        def _op(conn: Connection) -> Dict[str, Any]:
            with conn.cursor() as cur:
                cur.execute("""
                    SELECT name, state, failure_count, opened_at
                    FROM circuit_breakers
                    WHERE name = %s
                """, (name,))
                row = cur.fetchone()
                if not row:
                    return {"name": name, "state": "CLOSED", "failure_count": 0, "opened_at": None}
                return dict(row)

        return self.db.run_tx("cb_load", _op, settings=TxSettings(), trace_id=trace_id)

    def save(self, name: str, state: str, failure_count: int, opened_at: Optional[float], trace_id: str) -> None:
        def _op(conn: Connection) -> None:
            with conn.cursor() as cur:
                cur.execute("""
                    INSERT INTO circuit_breakers(name, state, failure_count, opened_at)
                    VALUES (%s, %s, %s, to_timestamp(%s))
                    ON CONFLICT (name) DO UPDATE SET
                        state = EXCLUDED.state,
                        failure_count = EXCLUDED.failure_count,
                        opened_at = EXCLUDED.opened_at
                """, (name, state, int(failure_count), opened_at if opened_at else None))
        self.db.run_tx("cb_save", _op, settings=TxSettings(), trace_id=trace_id)

@dataclass(frozen=True)
class SupervisorHealth:
    supervisor_id: str
    admin_status: str
    circuit_state: str
    opened_at_epoch: Optional[float]

    def is_operational(self) -> bool:
        return self.admin_status == "ACTIVE" and (self.circuit_state is None or self.circuit_state != "OPEN")

class SupervisorReconciler:
    def __init__(self, db: DB):
        self.db = db

    def reconcile(self, trace_id: str) -> Dict[str, int]:
        if os.environ.get(FAULT_INJECT_ENV) == "slow_reconcile":
            time.sleep(1.5)

        now = time.time()
        retired = 0
        spawned = 0

        def _op(conn: Connection) -> Dict[str, int]:
            nonlocal retired, spawned
            with conn.cursor() as cur:
                cur.execute("""
                    SELECT s.supervisor_id,
                           s.status AS admin_status,
                           cb.state AS circuit_state,
                           EXTRACT(EPOCH FROM cb.opened_at) AS opened_at_epoch
                    FROM supervisors s
                    LEFT JOIN circuit_breakers cb
                      ON cb.name = 'supervisor_' || s.supervisor_id
                    WHERE s.status = 'ACTIVE'
                    FOR UPDATE
                """)
                rows = cur.fetchall()

                health = [SupervisorHealth(
                    supervisor_id=r["supervisor_id"],
                    admin_status=r["admin_status"],
                    circuit_state=r.get("circuit_state"),
                    opened_at_epoch=r.get("opened_at_epoch"),
                ) for r in rows]

                operational = [h for h in health if h.is_operational()]

                SUPERVISOR_ACTIVE.set(len(health))
                SUPERVISOR_OPERATIONAL.set(len(operational))

                if len(operational) >= MIN_SUPERVISORS:
                    return {"retired": 0, "spawned": 0}

                dead = [
                    h for h in health
                    if h.circuit_state == "OPEN"
                    and h.opened_at_epoch is not None
                    and (now - float(h.opened_at_epoch)) > SUPERVISOR_CB_GRACE_PERIOD_SEC
                ]

                # Retire dead supervisors
                for d in dead:
                    cur.execute("""
                        UPDATE supervisors
                        SET status = 'RETIRED',
                            retired_at = NOW()
                        WHERE supervisor_id = %s
                    """, (d.supervisor_id,))
                    retired += 1

                # Recompute operational after retire intentions:
                new_operational_est = max(0, len(operational) - len(dead))
                needed = max(0, MIN_SUPERVISORS - new_operational_est)

                for _ in range(needed):
                    cur.execute("""
                        INSERT INTO supervisor_spawn_queue (spawn_id, reason, created_at)
                        VALUES (%s, 'auto_reconcile', NOW())
                    """, (f"spawn_{uuid.uuid4().hex}",))
                    spawned += 1

                if retired:
                    SUPERVISOR_RETIRED.inc(retired)
                if spawned:
                    SUPERVISOR_SPAWNED.inc(spawned)

                return {"retired": retired, "spawned": spawned}

        return self.db.run_tx("reconcile_supervisors", _op, settings=TxSettings(), trace_id=trace_id)

# ============================================================

class CredentialStore:
    def __init__(self, db: DB, keyring: Keyring):
        self.db = db
        self.keyring = keyring

    def put(self, name: str, secret: str, trace_id: str) -> str:
        cred_id = f"cred_{uuid.uuid4().hex}"
        aad = KEY_AAD_LABEL + b"|" + name.encode()

        blob = self.keyring.encrypt(secret.encode(), aad)

        def _op(conn: Connection) -> str:
            with conn.cursor() as cur:
                cur.execute("""
                    INSERT INTO credentials (cred_id, name, secret_enc, created_at)
                    VALUES (%s, %s, %s, NOW())
                """, (cred_id, name, blob))
            return cred_id

        return self.db.run_tx("cred_put", _op, settings=TxSettings(), trace_id=trace_id)

    def get(self, name: str, trace_id: str) -> str:
        aad = KEY_AAD_LABEL + b"|" + name.encode()

        def _op(conn: Connection) -> str:
            with conn.cursor() as cur:
                cur.execute("SELECT secret_enc FROM credentials WHERE name = %s", (name,))
                row = cur.fetchone()
                if not row:
                    raise KeyError("cred_not_found")
                secret_enc: bytes = row["secret_enc"]
                return self.keyring.decrypt(secret_enc, aad).decode()

        return self.db.run_tx("cred_get", _op, settings=TxSettings(), trace_id=trace_id)

    def rotate_reencrypt(self, *, batch: int, trace_id: str) -> int:
        """
        Online rotation helper:
        - Reads secrets (decrypt with old key id in envelope)
        - Re-encrypts with active key
        """
        def _op(conn: Connection) -> int:
            updated = 0
            with conn.cursor() as cur:
                cur.execute("""
                    SELECT cred_id, name, secret_enc
                    FROM credentials
                    ORDER BY created_at ASC
                    LIMIT %s
                    FOR UPDATE SKIP LOCKED
                """, (int(batch),))
                rows = cur.fetchall()

                for r in rows:
                    name = r["name"]
                    aad = KEY_AAD_LABEL + b"|" + name.encode()
                    plain = self.keyring.decrypt(r["secret_enc"], aad)
                    new_blob = self.keyring.encrypt(plain, aad)
                    cur.execute("""
                        UPDATE credentials
                        SET secret_enc = %s
                        WHERE cred_id = %s
                    """, (new_blob, r["cred_id"]))
                    updated += 1
            return updated

        return self.db.run_tx("cred_rotate", _op, settings=TxSettings(), trace_id=trace_id)

# ORCHESTRATOR (GRACEFUL SHUTDOWN + NON-DAEMON THREADS)
# ============================================================

class GracefulRunner:
    def __init__(self):
        self.stop = threading.Event()
        self.threads: List[threading.Thread] = []

    def start_thread(self, name: str, target: Callable[[], None]) -> None:
        t = threading.Thread(name=name, target=target, daemon=False)
        t.start()
        self.threads.append(t)

    def shutdown(self, timeout_sec: float = 15.0) -> None:
        self.stop.set()
        deadline = time.time() + timeout_sec
        for t in self.threads:
            remaining = max(0.0, deadline - time.time())
            t.join(timeout=remaining)

class Orchestrator:
    def __init__(self, db: DB):
        self.db = db
        self.runner = GracefulRunner()
        self.elector = LeaderElector(db)
        self.reconciler = SupervisorReconciler(db)
        self.metrics = MetricsCollector(db)
        self.adapter_registry = AdapterRegistry()
        self.last_reconcile = 0.0

    def start(self) -> None:
        self.runner.start_thread("orchestrator_loop", self._loop)
        self.runner.start_thread("adapter_cleanup", self._adapter_cleanup_loop)

    def stop(self) -> None:
        self.runner.shutdown()

    def _adapter_cleanup_loop(self) -> None:
        while not self.runner.stop.is_set():
            removed = self.adapter_registry.cleanup()
            if removed:
                LOGGER.info("adapter_evicted", extra={"trace_id": "no-trace", "op": "adapter_cleanup"})
            time.sleep(ADAPTER_CLEANUP_INTERVAL_SEC)

    def _loop(self) -> None:
        while not self.runner.stop.is_set():
            trace_id = Correlation.new_trace_id()
            with TRACER.start_as_current_span("orchestrator.loop"):
                acquired = False
                try:
                    acquired = self.elector.try_acquire(trace_id)
                    if not acquired:
                        LEADER_GAUGE.set(0)
                        time.sleep(1)
                        continue

                    LEADER_GAUGE.set(1)

                    now = time.time()
                    if now - self.last_reconcile >= RECONCILE_INTERVAL_SEC:
                        self.reconciler.reconcile(trace_id=trace_id)
                        self.last_reconcile = now

                    # Collect telemetry
                    self.metrics.collect_queue_depth(trace_id=trace_id)

                    # Task processing + outbox would go here (omitted: system-specific)
                    time.sleep(0.25)

                except Exception:
                    LOGGER.exception("orchestrator_loop_error", extra={"trace_id": trace_id, "op": "orchestrator_loop"})
                    time.sleep(1)
                finally:
                    if acquired:
                        try:
                            self.elector.release(trace_id)
                        except Exception:
                            LOGGER.exception("leader_release_failed", extra={"trace_id": trace_id, "op": "leader_release"})

def _fault_inject_check() -> None:
    if os.environ.get(FAULT_INJECT_ENV) == "db_fail":
        raise RuntimeError("fault_inject_db_fail")

def create_app() -> FastAPI:
    dsn = os.environ.get("DATABASE_DSN", "").strip()
    if not dsn:
        raise RuntimeError("Missing DATABASE_DSN")

    keyring = Keyring()
    db = DB(dsn)
    rate = RateLimiter(db)
    fam = FamilyRegistry(db)
    trans = TaskTransitions(db)
    creds = CredentialStore(db, keyring)
    health = HealthChecks(db)

    orch = Orchestrator(db)
    orch.start()

    app = FastAPI(title=SYSTEM_NAME, version=SYSTEM_VERSION)
    FastAPIInstrumentor.instrument_app(app)

    @app.middleware("http")
    async def metrics_mw(request: Request, call_next):
        path = request.url.path
        method = request.method
        start = time.time()
        try:
            resp = await call_next(request)
            code = str(resp.status_code)
            return resp
        finally:
            dur = max(0.0, time.time() - start)
            REQ_COUNT.labels(path=path, method=method, status=code if "code" in locals() else "500").inc()
            REQ_LAT.labels(path=path).observe(dur)

    def _trace_id_from_req(req: Request) -> str:
        incoming = req.headers.get("x-trace-id", "").strip()
        return incoming if incoming else Correlation.new_trace_id()

    def _rate_key(req: Request) -> str:
        # stable per ip + path
        ip = req.client.host if req.client else "unknown"
        return f"{ip}:{req.url.path}"

    @app.get("/metrics")
    def metrics_endpoint() -> Response:
        data = generate_latest()
        return Response(content=data, media_type=CONTENT_TYPE_LATEST)

    @app.get("/health")
    def health_endpoint(request: Request):
        trace_id = _trace_id_from_req(request)
        ok = health.db_ok(trace_id)
        return JSONResponse({"status": "healthy" if ok else "unhealthy", "db": ok})

    @app.get("/ready")
    def ready_endpoint(request: Request):
        trace_id = _trace_id_from_req(request)
        db_ok = health.db_ok(trace_id)
        schema_ok = health.schema_ok(trace_id)
        ready = bool(db_ok and schema_ok)
        return JSONResponse({"ready": ready, "db": db_ok, "schema": schema_ok})

    @app.post("/family/create")
    def create_family(request: Request, cfg: FamilyConfigIn):
        trace_id = _trace_id_from_req(request)
        if not rate.allow(_rate_key(request), tps=RATE_LIMIT_TOKENS_PER_SEC, burst=RATE_LIMIT_BURST, trace_id=trace_id):
            raise HTTPException(status_code=429, detail="rate_limited")

        _fault_inject_check()

        with TRACER.start_as_current_span("family.create"):
            family_id = fam.create_or_get(FamilyConfig(cfg.hunter_count, cfg.supervisor_layer), trace_id=trace_id)
            return JSONResponse({"family_id": family_id, "trace_id": trace_id})

    @app.post("/task/transition")
    def transition_task(request: Request, payload: TransitionIn):
        trace_id = _trace_id_from_req(request)
        if not rate.allow(_rate_key(request), tps=RATE_LIMIT_TOKENS_PER_SEC, burst=RATE_LIMIT_BURST, trace_id=trace_id):
            raise HTTPException(status_code=429, detail="rate_limited")

        with TRACER.start_as_current_span("task.transition"):
            try:
                trans.transition(payload.task_id, payload.to_stage, trace_id=trace_id)
            except TaskNotFoundError:
                raise HTTPException(status_code=404, detail="task_not_found")
            return JSONResponse({"ok": True, "trace_id": trace_id})

    @app.post("/credentials/put")
    def put_cred(request: Request, payload: CredentialIn):
        trace_id = _trace_id_from_req(request)
        if not rate.allow(_rate_key(request), tps=RATE_LIMIT_TOKENS_PER_SEC, burst=RATE_LIMIT_BURST, trace_id=trace_id):
            raise HTTPException(status_code=429, detail="rate_limited")

        with TRACER.start_as_current_span("credentials.put"):
            cred_id = creds.put(payload.name, payload.secret, trace_id=trace_id)
            return JSONResponse({"cred_id": cred_id, "trace_id": trace_id})

    @app.post("/credentials/get")
    def get_cred(request: Request, name: str):
        trace_id = _trace_id_from_req(request)
        if not rate.allow(_rate_key(request), tps=RATE_LIMIT_TOKENS_PER_SEC, burst=RATE_LIMIT_BURST, trace_id=trace_id):
            raise HTTPException(status_code=429, detail="rate_limited")

        with TRACER.start_as_current_span("credentials.get"):
            try:
                secret = creds.get(name, trace_id=trace_id)
            except KeyError:
                raise HTTPException(status_code=404, detail="cred_not_found")
            return JSONResponse({"name": name, "secret": secret, "trace_id": trace_id})

    @app.post("/credentials/rotate")
    def rotate_creds(request: Request, batch: int = 100):
        trace_id = _trace_id_from_req(request)
        if not rate.allow(_rate_key(request), tps=0.1, burst=1, trace_id=trace_id):
            raise HTTPException(status_code=429, detail="rate_limited")

        with TRACER.start_as_current_span("credentials.rotate"):
            updated = creds.rotate_reencrypt(batch=int(batch), trace_id=trace_id)
            return JSONResponse({"updated": updated, "trace_id": trace_id})

    @app.post("/admin/shutdown")
    def shutdown(request: Request):
        # For controlled environments only; protect in real deployment
        trace_id = _trace_id_from_req(request)
        orch.stop()
        db.close()
        return JSONResponse({"ok": True, "trace_id": trace_id})

    return app
