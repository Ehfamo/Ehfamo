# ============================================================
# DB ABSTRACTION (REQUIRED CONTRACT)
# ============================================================

class Database:
    def execute(self, sql: str, params: tuple = ()): ...
    def fetch_one(self, sql: str, params: tuple = ()): ...
    def fetch_all(self, sql: str, params: tuple = ()): ...

    @contextmanager
    def transaction(self):
        cur = self.execute("BEGIN")
        try:
            yield cur
            self.execute("COMMIT")
        except Exception:
            self.execute("ROLLBACK")
            raise

# ============================================================
# THREAD SAFE SINGLETON REGISTRY
# ============================================================

class Singleton:
    _instances: Dict[str, Any] = {}
    _lock = threading.Lock()

    @classmethod
    def get(cls, key: str, factory):
        with cls._lock:
            if key not in cls._instances:
                cls._instances[key] = factory()
            return cls._instances[key]

# FAMILY CREATION — DISTRIBUTED UNIQUENESS

@dataclass(frozen=True)
class FamilyConfig:
    hunter_count: int
    supervisor_layer: int

    def hash(self) -> str:
        return hashlib.sha256(
            json.dumps(self.__dict__, sort_keys=True).encode()
        ).hexdigest()

class FamilyRegistry:
    def __init__(self, db: Database):
        self.db = db

    def create(self, config: FamilyConfig) -> str:
        family_id = f"family_{uuid.uuid4().hex}"
        h = config.hash()

        with self.db.transaction():
            self.db.execute("""
                INSERT INTO families (family_id, config_hash, config, status)
                VALUES (%s, %s, %s::jsonb, 'ACTIVE')
                ON CONFLICT (config_hash) DO NOTHING
            """, (family_id, h, json.dumps(config.__dict__)))

            row = self.db.fetch_one(
                "SELECT family_id FROM families WHERE config_hash = %s",
                (h,)
            )
            return row[0]

class EvidenceStore:
    def __init__(self, db: Database):
        self.db = db

    def persist(self, task_id: str, payload: Dict[str, Any]):
        with self.db.transaction():
            exists = self.db.fetch_one(
                "SELECT 1 FROM task_queue WHERE task_id = %s",
                (task_id,)
            )
            if not exists:
                raise RuntimeError("TASK_NOT_FOUND")

            self.db.execute("""
                INSERT INTO evidence_bundles (evidence_id, task_id, payload)
                VALUES (%s, %s, %s::jsonb)
            """, (f"ev_{uuid.uuid4().hex}", task_id, json.dumps(payload)))

@dataclass
class ConversationContext:

    def add(self, msg: Dict[str, str], db: Database):
        self.messages.append(msg)

        if len(self.messages) > ARCHIVE_AT:
            old = self.messages[:-MAX_CONTEXT]
            self.messages = self.messages[-MAX_CONTEXT:]

            db.execute("""
                INSERT INTO conversation_archive (conversation_id, payload)

            """, (self.conversation_id, json.dumps(old)))

class LeaderElector:
    def __init__(self, db: Database):
        self.db = db

    def acquire(self) -> bool:
        return bool(self.db.fetch_one(
            "SELECT pg_try_advisory_lock(99999)"
        )[0])

    def release(self):
        self.db.execute("SELECT pg_advisory_unlock(99999)")

                SELECT s.supervisor_id, s.status, cb.state, cb.opened_at

                self.db.execute("""
                    UPDATE supervisors
                    SET status = 'RETIRED', retired_at = NOW()
