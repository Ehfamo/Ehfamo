@dataclass(frozen=True)
class SLO:
    name: str
    target: float
    window_seconds: int

TASK_SUCCESS_SLO = SLO(
    name="task_success_rate",
    target=0.999,          # 99.9%
    window_seconds=3600,   # 1 hour
)

TASK_LATENCY_SLO = SLO(
    name="task_latency_p99",
    target=10.0,           # seconds
    window_seconds=3600,
)

@dataclass
class SLOViolation:
    slo: SLO
    observed_value: float
    timestamp: float

class SLOEvaluator:
    """
    Single source of truth for SLA/SLO evaluation.
    """

    def __init__(self):
        self.violations: List[SLOViolation] = []

    def evaluate_success_rate(self, success: int, total: int):
        rate = success / max(total, 1)
        if rate < TASK_SUCCESS_SLO.target:
            self.violations.append(
                SLOViolation(TASK_SUCCESS_SLO, rate, time.time())
            )

    def evaluate_latency(self, p99_latency: float):
        if p99_latency > TASK_LATENCY_SLO.target:
            self.violations.append(
                SLOViolation(TASK_LATENCY_SLO, p99_latency, time.time())
            )

class ChaosMonkey:
    """
    Automated fault injection.
    Runs ONLY in non-production or controlled chaos mode.
    """

    def __init__(self, orchestrator, enabled: bool = False):
        self.orchestrator = orchestrator
        self.enabled = enabled

    def inject(self):
        if not self.enabled:
            return

        choice = random.choice([
            self._kill_orchestrator,
            self._kill_supervisor,
            self._simulate_db_latency,
        ])

        choice()

    def _kill_orchestrator(self):
        self.orchestrator.simulate_crash()

    def _kill_supervisor(self):
        self.orchestrator.kill_random_supervisor()

    def _simulate_db_latency(self):
        self.orchestrator.simulate_db_latency(seconds=2)

class ChaosRunner(threading.Thread):
    def __init__(self, chaos: ChaosMonkey, interval: int = 120):
        super().__init__(daemon=False)
        self.chaos = chaos
        self.interval = interval
        self._stop = threading.Event()

    def run(self):
        while not self._stop.is_set():
            self.chaos.inject()
            time.sleep(self.interval)

    def stop(self):
        self._stop.set()

TaskStatus = Literal["PENDING", "RUNNING", "COMPLETED", "FAILED"]
FailureType = Literal["ORCHESTRATOR", "SUPERVISOR", "DATABASE"]

@dataclass(frozen=True)
class TaskResult:
    task_id: str
    status: TaskStatus
    quality_score: float
    completed_at: float

class OrchestratorContract(Protocol):
    """
    This is the ONLY interface integration tests and chaos tools depend on.
    Internal refactors MUST NOT break this contract.
    """

    def submit_task(self, payload: Dict[str, Any]) -> str:
        """Submit a new task and return task_id"""
        ...

    def wait_for_completion(self, task_id: str, timeout_seconds: int = 300) -> TaskResult:
        """Block until task completes or timeout"""
        ...

    def simulate_failure(self, kind: FailureType) -> None:
        """Inject controlled failure (used by chaos tests)"""
        ...

    def operational_supervisor_count(self) -> int:
        """Return number of operational supervisors"""
        ...
