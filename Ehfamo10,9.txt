```python
@dataclass(frozen=True)
class EvidenceBundle:
    """
    Multi-dimensional evidence for quality assessment.
    Quality is NEVER a single number.
    Quality is derived from comprehensive evidence across dimensions.
    """
    evidence_id: str
    task_id: str
    family_id: str
    stage: Stage
    
    # DIMENSION 1: Content Quality Signals
    content_completeness: float  # 0-1: All required content present
    content_accuracy: float  # 0-1: Information verified
    content_format_validity: float  # 0-1: Format matches spec
    content_language_quality: float  # 0-1: Grammar, spelling, clarity
    content_depth: float  # 0-1: Sufficient detail
    
    # DIMENSION 2: Compliance Signals (Boolean)
    compliance_tos: bool  # Terms of service adherence
    compliance_legal: bool  # Legal requirements met
    compliance_policy: bool  # Internal policy adherence
    compliance_safety: bool  # Safety guidelines followed
    compliance_ethical: bool  # Ethical standards met
    
    # DIMENSION 3: Consistency Signals (Cross-Stage)
    consistency_semantic: float  # 0-1: Semantic alignment with previous stages
    consistency_data: float  # 0-1: Data integrity maintained
    consistency_contract: float  # 0-1: Deliverables match contract
    
    # DIMENSION 4: Delivery Signals
    delivery_usability: float  # 0-1: Client can use immediately
    delivery_satisfaction_proxy: float  # 0-1: Predicted satisfaction
    delivery_presentation: float  # 0-1: Professional presentation
    
    # Metadata
    collected_at: datetime = field(default_factory=datetime.now)
    collected_by: str = "qa_agent"
    evidence_version: str = "1.0.0"
    
    # Source data references
    source_artifacts: Dict[str, Any] = field(default_factory=dict)
    
    def __post_init__(self):
        """Validate evidence bundle"""
        # Validate all signals are 0-1
        float_signals = [
            self.content_completeness, self.content_accuracy,
            self.content_format_validity, self.content_language_quality,
            self.content_depth, self.consistency_semantic,
            self.consistency_data, self.consistency_contract,
            self.delivery_usability, self.delivery_satisfaction_proxy,
            self.delivery_presentation
        ]
        
        for signal in float_signals:
            if not (0 <= signal <= 1):
                raise ValueError(f"Signal out of range: {signal}")
        
        # Validate boolean signals are actually boolean
        bool_signals = [
            self.compliance_tos, self.compliance_legal,
            self.compliance_policy, self.compliance_safety,
            self.compliance_ethical
        ]
        
        for signal in bool_signals:
            if not isinstance(signal, bool):
                raise ValueError(f"Signal must be boolean: {signal}")
    
    def is_complete(self) -> bool:
        """Check if evidence bundle is complete"""
        # All signals must be present (not None)
        return all([
            self.content_completeness is not None,
            self.content_accuracy is not None,
            self.content_format_validity is not None,
            self.content_language_quality is not None,
            self.content_depth is not None,
            self.compliance_tos is not None,
            self.compliance_legal is not None,
            self.compliance_policy is not None,
            self.compliance_safety is not None,
            self.compliance_ethical is not None,
            self.consistency_semantic is not None,
            self.consistency_data is not None,
            self.consistency_contract is not None,
            self.delivery_usability is not None,
            self.delivery_satisfaction_proxy is not None,
            self.delivery_presentation is not None,
        ])
    
    def compute_quality_score(self) -> float:
        """
        Compute quality score using WEAKEST LINK principle.
        Quality = minimum across all dimensions.
        """
        # Content dimension: minimum of all content signals
        content_score = min([
            self.content_completeness,
            self.content_accuracy,
            self.content_format_validity,
            self.content_language_quality,
            self.content_depth
        ])
        
        # Compliance dimension: 1.0 if all True, 0.0 if any False
        compliance_score = 1.0 if all([
            self.compliance_tos,
            self.compliance_legal,
            self.compliance_policy,
            self.compliance_safety,
            self.compliance_ethical
        ]) else 0.0
        
        # Consistency dimension: minimum of consistency signals
        consistency_score = min([
            self.consistency_semantic,
            self.consistency_data,
            self.consistency_contract
        ])
        
        # Delivery dimension: minimum of delivery signals
        delivery_score = min([
            self.delivery_usability,
            self.delivery_satisfaction_proxy,
            self.delivery_presentation
        ])
        
        # Final quality: minimum across all dimensions (weakest link)
        final_quality = min([
            content_score,
            compliance_score,
            consistency_score,
            delivery_score
        ])
        
        return final_quality
    
    def meets_threshold(self) -> bool:
        """Check if evidence meets quality threshold"""
        return self.compute_quality_score() >= MIN_QUALITY_SCORE
    
    def get_failure_dimensions(self) -> List[str]:
        """Get dimensions that failed to meet threshold"""
        failures = []
        
        if min([self.content_completeness, self.content_accuracy,
                self.content_format_validity, self.content_language_quality,
                self.content_depth]) < MIN_QUALITY_SCORE:
            failures.append("content")
        
        if not all([self.compliance_tos, self.compliance_legal,
                    self.compliance_policy, self.compliance_safety,
                    self.compliance_ethical]):
            failures.append("compliance")
        
        if min([self.consistency_semantic, self.consistency_data,
                self.consistency_contract]) < MIN_QUALITY_SCORE:
            failures.append("consistency")
        
        if min([self.delivery_usability, self.delivery_satisfaction_proxy,
                self.delivery_presentation]) < MIN_QUALITY_SCORE:
            failures.append("delivery")
        
        return failures
```

```python
        if not (0 <= self.final_quality_score <= 1):

        # Verify meets_threshold matches computed value
        expected_meets = (self.final_quality_score >= self.threshold_used)
        if self.meets_threshold != expected_meets:
            raise ValueError("meets_threshold mismatch with computed value")
    
    def to_dict(self) -> Dict:
        """Convert to dictionary"""
        return {
            "certificate_id": self.certificate_id,
            "task_id": self.task_id,
            "family_id": self.family_id,
            "stage": self.stage.value,
            "evidence_bundle": asdict(self.evidence_bundle),
            "layer1_score": self.layer1_score,
            "layer2_score": self.layer2_score,
            "final_quality_score": self.final_quality_score,
            "meets_threshold": self.meets_threshold,
            "threshold_used": self.threshold_used,
            "issued_at": self.issued_at.isoformat(),
            "issued_by": self.issued_by,
```

```python
@dataclass
class DecisionNode:

    stage: Stage

    agent_type: str  # "hunter", "worker", "qa_agent", etc.

@dataclass
class DecisionTrace:
    """
    Complete decision trace for a task.
    Enables full forensic reconstruction.
    """
    trace_id: str
    task_id: str
    family_id: str
    
    # Decision chain
    nodes: List[DecisionNode] = field(default_factory=list)
    
    # Summary
    root_decision: str  # Initial decision that started trace
    final_outcome: Dict[str, Any]  # Final result
    total_confidence: float = 1.0
    
    # Metadata
    trace_started_at: datetime = field(default_factory=datetime.now)
    trace_completed_at: Optional[datetime] = None
    
    def add_node(self, node: DecisionNode):
        """Add decision node to trace"""
        self.nodes.append(node)
        # Update total confidence (multiply confidences)
        self.total_confidence *= node.confidence
    
    def get_nodes_by_stage(self, stage: Stage) -> List[DecisionNode]:
        """Get all nodes for a specific stage"""
        return [n for n in self.nodes if n.stage == stage]
    
    def get_nodes_by_type(self, decision_type: str) -> List[DecisionNode]:
        """Get all nodes of specific type"""
        return [n for n in self.nodes if n.decision_type == decision_type]
    
    def reconstruct_timeline(self) -> List[Dict]:
        """Reconstruct chronological timeline"""
        return [
            {
                "timestamp": node.timestamp.isoformat(),
                "stage": node.stage.value,
                "decision_type": node.decision_type,
                "agent": node.agent_id,
                "confidence": node.confidence,
                "output": node.output
            }
            for node in sorted(self.nodes, key=lambda n: n.timestamp)
        ]
    
    def explain(self) -> str:
        """Generate human-readable explanation"""
        explanation = f"Decision Trace for Task {self.task_id}\n"
        explanation += f"Started: {self.trace_started_at.isoformat()}\n"
        explanation += f"Total Confidence: {self.total_confidence:.4f}\n\n"
        
        for i, node in enumerate(self.nodes, 1):
            explanation += f"{i}. [{node.stage.value}] {node.decision_type}\n"
            explanation += f"   Agent: {node.agent_id} ({node.agent_type})\n"
            explanation += f"   Logic: {node.logic_applied}\n"
            explanation += f"   Confidence: {node.confidence:.4f}\n"
            explanation += f"   Output: {node.output}\n\n"
        
        return explanation
```

```python
import psycopg
from psycopg_pool import ConnectionPool
from typing import Optional, ContextManager
from contextlib import contextmanager
import logging

logger = logging.getLogger(__name__)

class DatabaseConnectionPool:
    """
    Production-grade connection pool with health checks and automatic recovery.
    """
    
    def __init__(
        self,
        dsn: str,
        min_size: int = 10,
        max_size: int = 50,
        timeout: float = 30.0,
        max_lifetime: float = 3600.0,
        max_idle: float = 600.0
    ):
        self.dsn = dsn
        self._pool: Optional[ConnectionPool] = None
        self.min_size = min_size
        self.max_size = max_size
        self.timeout = timeout
        self.max_lifetime = max_lifetime
        self.max_idle = max_idle
        
        self._initialize_pool()
    
    def _initialize_pool(self):
        """Initialize connection pool"""
        try:
            self._pool = ConnectionPool(
                self.dsn,
                min_size=self.min_size,
                max_size=self.max_size,
                timeout=self.timeout,
                max_lifetime=self.max_lifetime,
                max_idle=self.max_idle,
                open=True
            )
            logger.info("database_pool_initialized", extra={
                "min_size": self.min_size,
                "max_size": self.max_size
            })
        except Exception as e:
            logger.critical("database_pool_init_failed", extra={"error": str(e)})
            raise
    
    @contextmanager
    def connection(self) -> ContextManager[psycopg.Connection]:
        """Get connection from pool with automatic return"""
        if not self._pool:
            raise RuntimeError("Connection pool not initialized")
        
        conn = None
        try:
            conn = self._pool.getconn(timeout=self.timeout)
            
            # Health check
            conn.execute("SELECT 1")
            
            yield conn
            
        except Exception as e:
            logger.error("database_connection_error", extra={"error": str(e)})
            if conn:
                try:
                    conn.rollback()
                except:
                    pass
            raise
        finally:
            if conn:
                self._pool.putconn(conn)
    
    @contextmanager
    def transaction(self) -> ContextManager[psycopg.Connection]:
        """Get connection with automatic transaction management"""
        with self.connection() as conn:
            try:
                with conn.transaction():
                    yield conn
            except Exception:
                # Transaction automatically rolled back
                raise
    
    def close(self):
        """Close connection pool"""
        if self._pool:
            self._pool.close()
            logger.info("database_pool_closed")
    
    def get_stats(self) -> dict:
        """Get pool statistics"""
        if not self._pool:
            return {}
        
        return {
            "size": self._pool.get_stats().get("pool_size", 0),
            "available": self._pool.get_stats().get("pool_available", 0),
            "waiting": self._pool.get_stats().get("requests_waiting", 0),
        }
```

```python
class SignalAnalyzer:
    """
    Analyzes raw signals to extract intelligence.
    Pure CPU-based analysis using rules and statistics.
    """
    
    def __init__(self):
        self.min_confidence = 0.6
        self.clustering_threshold = 0.7
    
    def analyze_signal(self, signal: MarketSignal) -> Optional[MarketIntelligence]:
        """
        Analyze single signal.
        Returns intelligence if signal is actionable.
        """
        if signal.confidence < self.min_confidence:
            logger.debug("signal_low_confidence", extra={
                "signal_id": signal.signal_id,
                "confidence": signal.confidence
            })
            return None
        
        # Route to appropriate analyzer
        if signal.source == SignalSource.FREELANCE_PLATFORM_TRENDS:
            return self._analyze_freelance_trend(signal)
        elif signal.source == SignalSource.JOB_POSTING_CLUSTERS:
            return self._analyze_job_cluster(signal)
        elif signal.source == SignalSource.COMPETITION_GAPS:
            return self._analyze_competition_gap(signal)
        elif signal.source == SignalSource.SEARCH_VOLUME_DATA:
            return self._analyze_search_volume(signal)
        else:
            return self._analyze_generic(signal)
    
    def _analyze_freelance_trend(self, signal: MarketSignal) -> Optional[MarketIntelligence]:
        """Analyze freelance platform trend"""
        data = signal.raw_data
        
        # Extract key metrics
        keyword = data.get("keyword", "")
        volume = data.get("volume", 0)
        trend = data.get("trend", "stable")
        avg_price = data.get("avg_price", 0)
        competition = data.get("competition", 0.5)
        
        # Scoring logic
        opportunity_score = self._compute_opportunity_score(
            volume=volume,
            trend_direction=trend,
            avg_price=avg_price,
            competition=competition
        )
        
        if opportunity_score < 0.6:
            return None  # Not interesting enough
        
        # Estimate market size
        market_size = self._estimate_market_size(volume, avg_price)
        
        analyzed_data = {
            "keyword": keyword,
            "opportunity_score": opportunity_score,
            "volume": volume,
            "trend_direction": trend,
            "avg_price": avg_price,
            "competition_level": competition,
            "estimated_monthly_revenue": market_size / 12,
            "viability": "high" if opportunity_score > 0.8 else "medium"
        }
        
        return MarketIntelligence(
            intelligence_id=f"intel_{uuid.uuid4().hex}",
            intelligence_type=IntelligenceType.DEMAND_SIGNAL,
            raw_data=signal.raw_data,
            analyzed_data=analyzed_data,
            confidence_score=signal.confidence * opportunity_score,
            industry=signal.industry or self._classify_industry(keyword),
            geography=signal.geography,
            market_size_estimate=market_size,
            status="DETECTED",
            discovered_at=signal.detected_at
        )
    
    def _analyze_job_cluster(self, signal: MarketSignal) -> Optional[MarketIntelligence]:
        """Analyze job posting cluster"""
        data = signal.raw_data
        
        cluster_size = data.get("cluster_size", 0)
        common_keywords = data.get("common_keywords", [])
        avg_budget = data.get("avg_budget", 0)
        posting_frequency = data.get("posting_frequency", 0)
        
        if cluster_size < 50:  # Minimum cluster size
            return None
        
        # Calculate cluster density (jobs per day)
        density = posting_frequency / max(cluster_size, 1)
        
        # Score based on density and budget
        cluster_score = min(1.0, (density * avg_budget) / 10000)
        
        if cluster_score < 0.5:
            return None
        
        analyzed_data = {
            "cluster_size": cluster_size,
            "keywords": common_keywords,
            "avg_budget": avg_budget,
            "posting_density": density,
            "cluster_score": cluster_score,
            "industrializable": cluster_score > 0.7
        }
        
        market_size = cluster_size * avg_budget * 12  # Annual estimate
        
        return MarketIntelligence(

            intelligence_type=IntelligenceType.OPPORTUNITY_CLUSTER,
            raw_data=signal.raw_data,

    def _analyze_competition_gap(self, signal: MarketSignal) -> Optional[MarketIntelligence]:
        """Analyze competition gap"""
        data = signal.raw_data
        
        demand_level = data.get("demand_level", 0)
        supply_level = data.get("supply_level", 0)
        gap_size = data.get("gap_size", 0)
        
        # Gap must be significant
        if gap_size < 0.3:

        # Demand must be substantial
        if demand_level < 100:
            return None
        
        gap_score = min(1.0, gap_size * (demand_level / 1000))
        
        analyzed_data = {
            "demand_level": demand_level,
            "supply_level": supply_level,
            "gap_size": gap_size,
            "gap_score": gap_score,
            "exploitability": "high" if gap_score > 0.7 else "medium"
        }
        
        market_size = demand_level * data.get("avg_value", 100) * 12

    def _analyze_search_volume(self, signal: MarketSignal) -> Optional[MarketIntelligence]:
        """Analyze search volume trends"""
        data = signal.raw_data
        
        keyword = data.get("keyword", "")
        monthly_volume = data.get("monthly_volume", 0)
        growth_rate = data.get("growth_rate", 0)
        cpc = data.get("cpc", 0)
        
        # Must have meaningful volume
        if monthly_volume < 1000:
            return None
        
        # Positive growth preferred
        if growth_rate < -0.1:
            return None
        
        trend_score = min(1.0, (monthly_volume / 10000) * (1 + growth_rate))
        
        analyzed_data = {
            "keyword": keyword,
            "monthly_volume": monthly_volume,
            "growth_rate": growth_rate,
            "cpc": cpc,
            "trend_score": trend_score,
            "trend_strength": "strong" if growth_rate > 0.2 else "moderate"
        }
        
        # Estimate market size from search volume and CPC
        market_size = monthly_volume * cpc * 0.05 * 12  # 5% conversion estimate
        
        return MarketIntelligence(
            intelligence_id=f"intel_{uuid.uuid4().hex}",
            intelligence_type=IntelligenceType.TREND,
            raw_data=signal.raw_data,
            analyzed_data=analyzed_data,
            confidence_score=signal.confidence * trend_score,
            industry=self._classify_industry(keyword),
            geography=signal.geography,
            market_size_estimate=market_size,
            status="DETECTED",
            discovered_at=signal.detected_at
        )

            geography=signal.geography,
            market_size_estimate=None,
            status="DETECTED",
            discovered_at=signal.detected_at
        )
    
    def _compute_opportunity_score(
        self,
        volume: int,
        trend_direction: str,
        avg_price: float,
        competition: float
    ) -> float:
        """Compute opportunity score (0-1)"""
        # Volume score (normalized to 0-1)
        volume_score = min(1.0, volume / 5000)
        
        # Trend multiplier
        trend_multiplier = {
            "increasing": 1.2,
            "stable": 1.0,
            "decreasing": 0.7
        }.get(trend_direction, 1.0)
        
        # Price score (normalized)
        price_score = min(1.0, avg_price / 500)
        
        # Competition penalty (inverse)
        competition_score = 1.0 - competition
        
        # Combined score
        raw_score = (
            0.3 * volume_score +
            0.2 * price_score +
            0.5 * competition_score
        ) * trend_multiplier
        
        return min(1.0, max(0.0, raw_score))
    
    def _estimate_market_size(self, volume: int, avg_price: float) -> float:
        """Estimate annual market size"""
        # Conservative estimate: 10% of volume converts
        conversion_rate = 0.1
        return volume * avg_price * conversion_rate * 12
    
    def _classify_industry(self, keyword: str) -> str:
        """Classify industry from keyword"""
        keyword_lower = keyword.lower()
        
        if any(w in keyword_lower for w in ["write", "content", "blog", "article"]):
            return "content"
        elif any(w in keyword_lower for w in ["seo", "search", "ranking"]):
            return "seo"
        elif any(w in keyword_lower for w in ["design", "logo", "graphic"]):
            return "design"
        elif any(w in keyword_lower for w in ["code", "develop", "program", "software"]):
            return "development"
        elif any(w in keyword_lower for w in ["data", "analysis", "research"]):
            return "data"
        elif any(w in keyword_lower for w in ["market", "social", "ads"]):
            return "marketing"
        else:
            return "general"
    
    def _classify_industry_from_keywords(self, keywords: List[str]) -> str:
        """Classify industry from list of keywords"""
        if not keywords:
            return "general"
        
        # Use first keyword for classification
        return self._classify_industry(keywords[0])
```

```python
class IdeationEngine:
    """
    Generates family creation proposals from intelligence.
    MANDATORY: All proposals require evidence.
    MANDATORY: All proposals subject to AI quorum approval.
    """
    
    def __init__(self):
        self.min_confidence = 0.7
        self.min_market_size = 50000  # $50k annual minimum
        self.cooldown_hours = 24  # Between similar proposals
    
    def evaluate_intelligence_for_ideation(
        self,
        intelligence: MarketIntelligence
    ) -> bool:
        """Determine if intelligence warrants ideation"""
        
        # Confidence gate
        if intelligence.confidence_score < self.min_confidence:
            logger.debug("intelligence_low_confidence", extra={
                "intelligence_id": intelligence.intelligence_id
            })
            return False
        
        # Market size gate
        if intelligence.market_size_estimate:
            if intelligence.market_size_estimate < self.min_market_size:
                logger.debug("intelligence_small_market", extra={
                    "intelligence_id": intelligence.intelligence_id,
                    "market_size": intelligence.market_size_estimate
                })
                return False
        
        # Status gate
        if intelligence.status != "VALIDATED":
            return False
        
        # Cooldown check
        if self._is_in_cooldown(intelligence):
            return False
        
        return True
    
    def _is_in_cooldown(self, intelligence: MarketIntelligence) -> bool:
        """Check if similar proposals were recently made"""
        db = get_db()
        with db.connection() as conn:
            cursor = conn.execute("""
                SELECT COUNT(*) FROM ideation_proposals
                WHERE proposed_family_type = %s
                  AND created_at > NOW() - INTERVAL '%s hours'
            """, (intelligence.industry, self.cooldown_hours))
            
            count = cursor.fetchone()[0]
            return count > 0
    
    def generate_proposal(
        self,
        intelligence: MarketIntelligence
    ) -> 'IdeationProposal':
        """
        Generate family creation proposal from intelligence.
        """
        
        # Extract key characteristics

        # Estimate metrics
        opportunity_count = self._estimate_opportunity_count(intelligence)

        )
        
        return proposal

        type_mapping = {
            "content": "content_freelance",
            "seo": "seo_services",
            "design": "design_freelance",
            "development": "dev_freelance",
            "data": "data_services",
            "marketing": "marketing_services",
            "general": "general_freelance"
        }
        
        return type_mapping.get(industry, "general_freelance")
    
    def _generate_family_name(
        self,

    ) -> str:
        """Generate descriptive family name"""

        geography = intelligence.geography or "global"

        rationale_parts = [

            return int(analyzed["cluster_size"] * 0.1)
        elif "monthly_volume" in analyzed:
            # Search volume to opportunities
            return int(analyzed["monthly_volume"] * 0.001)

        """Estimate initial investment required"""
        # Base costs

        type_multipliers = {
            "content_freelance": 1.0,
            "seo_services": 1.5,
            "design_freelance": 1.2,
            "dev_freelance": 2.0,
            "data_services": 1.8,
            "marketing_services": 1.3,
            "general_freelance": 1.0
        }
        
        multiplier = type_multipliers.get(family_type, 1.0)
        
        monthly_investment = (base_hunter_cost + base_worker_cost + base_infrastructure) * multiplier
        
        # 3-month ramp-up period
        return monthly_investment * 3
    
    def _assess_risks(self, intelligence: MarketIntelligence) -> str:

        if "competition_level" in analyzed:

                risks.append("High competition - differentiation required")
        
        # Data completeness risk
        if not intelligence.market_size_estimate:
            risks.append("Limited market data - estimates uncertain")
        
        if not risks:
            return "Low risk - strong opportunity with good data"
        
        return "Risks: " + "; ".join(risks)
```

```python
        saturation_check = self._check_market_saturation(proposal)
        
        evaluation = {
            "proposal_id": proposal.proposal_id,
            "viability_score": viability_score,
            "market_assessment": market_assessment,
            "financial_assessment": financial_assessment,
            "risk_score": risk_score,
            "saturation_check": saturation_check,
            "recommendation": "APPROVE" if viability_score >= self.min_viability_score else "REJECT",
            "evaluated_at": datetime.now().isoformat()
        }
        
        return evaluation
    
    def _compute_viability_score(self, proposal: IdeationProposal) -> float:
        """Compute overall viability (0-1)"""
        
        # Market size factor
        if proposal.estimated_revenue_potential:
            revenue_score = min(1.0, proposal.estimated_revenue_potential / 500000)
        else:
            revenue_score = 0.3
        
        # Opportunity volume factor
        if proposal.estimated_opportunity_count:
            volume_score = min(1.0, proposal.estimated_opportunity_count / 1000)
        else:
            volume_score = 0.3
        
        # ROI factor
        if proposal.estimated_revenue_potential and proposal.estimated_required_investment:
            roi = proposal.estimated_revenue_potential / proposal.estimated_required_investment
            roi_score = min(1.0, roi / 10)
        else:
            roi_score = 0.5
        
        # Evidence quality factor
        evidence = proposal.evidence
        evidence_score = evidence.get("confidence_score", 0.5)
        
        # Combined viability
        viability = (
            0.3 * revenue_score +
            0.2 * volume_score +
            0.3 * roi_score +
            0.2 * evidence_score
        )
        
        return viability
    
    def _assess_market(self, proposal: IdeationProposal) -> Dict:
        """Assess market characteristics"""
        evidence = proposal.evidence
        
        return {
            "market_size": proposal.estimated_revenue_potential,
            "opportunity_volume": proposal.estimated_opportunity_count,
            "industry": evidence.get("industry"),
            "geography": evidence.get("geography"),
            "growth_potential": "high" if proposal.estimated_revenue_potential and proposal.estimated_revenue_potential > 200000 else "medium"
        }
    
    def _assess_financials(self, proposal: IdeationProposal) -> Dict:
        """Assess financial viability"""
        revenue = proposal.estimated_revenue_potential or 0
        investment = proposal.estimated_required_investment or 1
        
        roi = revenue / investment if investment > 0 else 0
        breakeven_months = investment / (revenue / 12) if revenue > 0 else 999
        
        return {
            "estimated_annual_revenue": revenue,
            "required_investment": investment,
            "estimated_roi": roi,
            "breakeven_months": min(breakeven_months, 999),
            "financial_viability": "strong" if roi > 5 else "acceptable" if roi > 2 else "weak"
        }
    
    def _compute_risk_score(self, proposal: IdeationProposal) -> float:
        """Compute risk score (0-1, higher = riskier)"""
        risk_factors = []
        
        # Evidence confidence risk
        evidence = proposal.evidence
        confidence = evidence.get("confidence_score", 0.5)
        risk_factors.append(1.0 - confidence)
        
        # Market size risk
        if proposal.estimated_revenue_potential:
            if proposal.estimated_revenue_potential < 100000:
                risk_factors.append(0.7)
            else:
                risk_factors.append(0.2)
        else:
            risk_factors.append(0.9)
        
        # Investment risk
        if proposal.estimated_required_investment:
            if proposal.estimated_required_investment > 50000:
                risk_factors.append(0.6)
            else:
                risk_factors.append(0.3)
        else:
            risk_factors.append(0.5)
        
        return sum(risk_factors) / len(risk_factors)
    
    def _check_market_saturation(self, proposal: IdeationProposal) -> Dict:
        """Check if market is already saturated by existing families"""
        db = get_db()
        
        with db.connection() as conn:
            cursor = conn.execute("""
                SELECT COUNT(*) FROM families
                WHERE family_type = %s AND status = 'ACTIVE'
            """, (proposal.proposed_family_type,))
            
            existing_count = cursor.fetchone()[0]
        
        saturated = existing_count >= 3  # Max 3 families per type
        
        return {
            "existing_families": existing_count,
            "saturated": saturated,
            "can_add": not saturated
        }

def submit_proposal_for_approval(proposal: IdeationProposal) -> bool:
```

```python
class SupervisorLifecycleManager:
    """Manages supervisor creation, activation, suspension, retirement"""
    
    def __init__(self):
        self.heartbeat_timeout_seconds = 300  # 5 minutes
        self.replacement_delay_seconds = 60  # 1 minute
    
    def create_supervisor(
        self,
        layer: SupervisorLayer,
        family_id: Optional[str] = None,
        domain: Optional[str] = None
    ) -> Supervisor:
        """Create new supervisor"""
        
        # Validate layer-specific requirements
        if layer == SupervisorLayer.LOCAL and not family_id:
            raise ValueError("Layer 1 requires family_id")
        if layer == SupervisorLayer.DOMAIN and not domain:
            raise ValueError("Layer 2 requires domain")
        if layer == SupervisorLayer.GLOBAL and (family_id or domain):
            raise ValueError("Layer 3 cannot have family_id or domain")
        
        # Generate name
        if layer == SupervisorLayer.LOCAL:
            name = f"Local-{family_id}"
        elif layer == SupervisorLayer.DOMAIN:
            name = f"Domain-{domain}"
        else:
            name = "Global-MetaGovernor"
        
        supervisor = Supervisor(
            supervisor_id=f"supervisor_{uuid.uuid4().hex}",
            supervisor_name=name,
            layer=layer,
            family_id=family_id,
            domain=domain,
            status=SupervisorStatus.ACTIVE,
            created_at=datetime.now(),
            activated_at=datetime.now()
        )
        
        # Persist
        self._persist_supervisor(supervisor)
        
        logger.info("supervisor_created", extra={
            "supervisor_id": supervisor.supervisor_id,
            "layer": layer.value,
            "family_id": family_id,
            "domain": domain
        })
        
        return supervisor
    
    def activate_supervisor(self, supervisor_id: str):
        """Activate suspended supervisor"""
        db = get_db()

            conn.execute("""
                UPDATE supervisors
                SET status = 'ACTIVE', activated_at = NOW(), suspended_at = NULL

        logger.info("supervisor_activated", extra={"supervisor_id": supervisor_id})

                UPDATE supervisors

                UPDATE supervisors

            """, (supervisor_id,))

                supervisor_id=supervisor_id,
                details={"reason": reason}

            "supervisor_id": supervisor_id,
            "reason": reason
        })

        db = get_db()

                UPDATE supervisors
                SET last_heartbeat = NOW()
                WHERE supervisor_id = %s
            """, (supervisor_id,))
    
    def check_heartbeats_and_replace(self):
        """Check for dead supervisors and replace them"""
        db = get_db()
        
        # Find dead supervisors
        timeout_threshold = datetime.now() - timedelta(seconds=self.heartbeat_timeout_seconds)
        
        with db.connection() as conn:
            cursor = conn.execute("""
                SELECT supervisor_id, layer, family_id, domain
                FROM supervisors
                WHERE status = 'ACTIVE'
                  AND (last_heartbeat < %s OR last_heartbeat IS NULL)
            """, (timeout_threshold,))
            
            dead_supervisors = cursor.fetchall()

            supervisor_id, layer, family_id, domain = row
            
            logger.critical("supervisor_heartbeat_timeout", extra={

                "layer": layer

            # Replace immediately for critical layers

                self.create_supervisor(
                    layer=SupervisorLayer(layer),
                    family_id=family_id,
                    domain=domain
                )
                
                logger.info("supervisor_replaced", extra={
                    "old_supervisor_id": supervisor_id,
                    "layer": layer
                })

                    status, interventions_count, escalations_count,
                    last_heartbeat, created_at
```

```python
@dataclass
class ScalingJustification:
    """Evidence-based justification for scaling"""
    justification_id: str
    target_type: str  # "supervisor", "hunter", "worker"
    family_id: Optional[str]

    current_count: int
    proposed_count: int

        if not self.sustained_signal:
            return False
        if self.signal_duration_hours < 24:
            return False
        return True

            "expected_throughput_impact": self.expected_throughput_impact,

class SupervisorScalingEngine:
    """
    Manages supervisor scaling with strict requirements:
    - Statistical confidence >= 0.95
    - AI quorum approval (3 judges)
    - 24h cooldown
    - Anti-noise (sustained signal)
    """
    
    def __init__(self):
        self.baseline_supervisor_count = 30
        self.cooldown_hours = 24
    
    def evaluate_scaling_need(self) -> Optional[ScalingJustification]:
        """
        Evaluate if supervisor scaling is needed.
        Returns justification if scaling warranted, None otherwise.
        """
        
        # Get current supervisor count
        current_count = self._get_active_supervisor_count()
        
        # Check cooldown
        if self._is_in_cooldown():
            logger.debug("supervisor_scaling_in_cooldown")
            return None
        
        # Gather metrics
        metrics = self._gather_scaling_metrics()
        
        # Compute need
        scaling_signal = self._compute_scaling_signal(metrics)
        
        if not scaling_signal["needs_scaling"]:
            return None
        
        proposed_count = scaling_signal["proposed_count"]
        
        # Enforce bounds
        if proposed_count < MIN_SUPERVISORS or proposed_count > MAX_SUPERVISORS:
            logger.warning("supervisor_scaling_out_of_bounds", extra={
                "proposed": proposed_count
            })
            return None
        
        # Check sustained signal
        sustained = self._check_sustained_signal(metrics)
        
        if not sustained["is_sustained"]:
            logger.debug("supervisor_scaling_signal_not_sustained")
            return None
        
        # Build justification
        justification = ScalingJustification(
            justification_id=f"justification_{uuid.uuid4().hex}",
            target_type="supervisor",
            family_id=None,  # System-wide
            current_count=current_count,
            proposed_count=proposed_count,
            delta=proposed_count - current_count,
            statistical_confidence=metrics["statistical_confidence"],
            data_points=metrics["data_points"],
            observation_window_hours=metrics["observation_window_hours"],
            supporting_metrics=metrics["raw_metrics"],
            expected_quality_impact=scaling_signal["expected_quality_impact"],
            expected_throughput_impact=scaling_signal["expected_throughput_impact"],
            expected_cost_impact=scaling_signal["expected_cost_impact"],
            sustained_signal=sustained["is_sustained"],
            signal_duration_hours=sustained["duration_hours"]
        )
        
        return justification
    
    def _get_active_supervisor_count(self) -> int:
        """Get current active supervisor count"""
        db = get_db()
        with db.connection() as conn:
            cursor = conn.execute("""
                SELECT COUNT(*) FROM supervisors WHERE status = 'ACTIVE'
            """)
            return cursor.fetchone()[0]
    
    def _is_in_cooldown(self) -> bool:
        """Check if system is in scaling cooldown"""
        db = get_db()
        with db.connection() as conn:
            cursor = conn.execute("""
                SELECT MAX(created_at) FROM scaling_history
                WHERE target_type = 'supervisor' AND executed = TRUE
            """)
            last_scaling = cursor.fetchone()[0]
        
        if not last_scaling:
            return False
        
        hours_since = (datetime.now() - last_scaling).total_seconds() / 3600
        return hours_since < self.cooldown_hours
    
    def _gather_scaling_metrics(self) -> Dict:
        """Gather metrics for scaling decision"""
        # Query metrics from last 48 hours
        metrics = query_metrics_bulk([
            "supervisor.interventions_per_hour",
            "supervisor.escalations_per_hour",
            "supervisor.avg_response_time_seconds",
            "system.total_pending_tasks",
            "system.quality_incidents_per_hour"
        ], window_hours=48)
        
        # Compute statistical confidence
        if not metrics or len(metrics["supervisor.interventions_per_hour"]) < 100:
            confidence = 0.0
        else:
            # Bootstrap confidence interval
            confidence = self._compute_bootstrap_confidence(
                metrics["supervisor.interventions_per_hour"]
            )
        
        return {
            "statistical_confidence": confidence,
            "data_points": len(metrics.get("supervisor.interventions_per_hour", [])),
            "observation_window_hours": 48,
            "raw_metrics": {
                "avg_interventions_per_hour": self._mean(metrics.get("supervisor.interventions_per_hour", [0])),
                "avg_escalations_per_hour": self._mean(metrics.get("supervisor.escalations_per_hour", [0])),
                "avg_response_time": self._mean(metrics.get("supervisor.avg_response_time_seconds", [0])),
                "avg_pending_tasks": self._mean(metrics.get("system.total_pending_tasks", [0])),
                "avg_quality_incidents": self._mean(metrics.get("system.quality_incidents_per_hour", [0]))
            }
        }
    
    def _compute_scaling_signal(self, metrics: Dict) -> Dict:
        """Compute scaling signal from metrics"""
        raw = metrics["raw_metrics"]
        current_count = self._get_active_supervisor_count()
        
        # Load per supervisor
        interventions_per_supervisor = raw["avg_interventions_per_hour"] / max(current_count, 1)
        
        # Thresholds
        overload_threshold = 10  # interventions per supervisor per hour
        underload_threshold = 2
        
        needs_scaling = False
        proposed_count = current_count
        
        if interventions_per_supervisor > overload_threshold:
            # Scale up
            scale_factor = interventions_per_supervisor / overload_threshold
            proposed_count = min(MAX_SUPERVISORS, int(current_count * scale_factor))
            needs_scaling = True
        elif interventions_per_supervisor < underload_threshold and current_count > self.baseline_supervisor_count:
            # Scale down (but not below baseline)
            scale_factor = interventions_per_supervisor / underload_threshold
            proposed_count = max(self.baseline_supervisor_count, int(current_count * scale_factor))
            needs_scaling = proposed_count != current_count
        
        # Expected impacts
        if needs_scaling:
            delta_ratio = (proposed_count - current_count) / current_count
            expected_quality_impact = 0.05 * delta_ratio  # 5% improvement per 100% increase
            expected_throughput_impact = 0.1 * delta_ratio  # 10% improvement per 100% increase
            expected_cost_impact = delta_ratio  # Linear cost
        else:
            expected_quality_impact = 0.0
            expected_throughput_impact = 0.0
            expected_cost_impact = 0.0
        
        return {
            "needs_scaling": needs_scaling,
            "proposed_count": proposed_count,
            "expected_quality_impact": expected_quality_impact,
            "expected_throughput_impact": expected_throughput_impact,
            "expected_cost_impact": expected_cost_impact
        }
    
    def _check_sustained_signal(self, metrics: Dict) -> Dict:
        """Check if scaling signal is sustained over time"""
        # Check if signal has been consistent for 24+ hours
        # Split data into 4-hour windows
        interventions = metrics["raw_metrics"]["avg_interventions_per_hour"]
        
        if not interventions:
            return {"is_sustained": False, "duration_hours": 0}
        
        # Simplified: check if signal sustained for full observation window
        duration_hours = metrics["observation_window_hours"]
        is_sustained = duration_hours >= 24
        
        return {
            "is_sustained": is_sustained,
            "duration_hours": duration_hours
        }
    
    def _compute_bootstrap_confidence(self, data: List[float]) -> float:
        """Compute bootstrap confidence interval"""
        if len(data) < 100:
            return 0.0
        
        import random
        
        # Bootstrap resampling
        n_bootstrap = 1000
        means = []
        
        for _ in range(n_bootstrap):
            sample = random.choices(data, k=len(data))
            means.append(sum(sample) / len(sample))
        
        # Confidence interval
        means.sort()
        lower = means[int(0.025 * n_bootstrap)]
        upper = means[int(0.975 * n_bootstrap)]
        
        # Convert to confidence score
        mean_val = sum(data) / len(data)
        if mean_val == 0:
            return 1.0
        
        interval_width = (upper - lower) / mean_val
        confidence = max(0.0, 1.0 - interval_width)
        
        return confidence
    
    def _mean(self, values: List[float]) -> float:
        """Compute mean"""
        if not values:
            return 0.0
        return sum(values) / len(values)
    
    def submit_scaling_for_approval(
        self,
        justification: ScalingJustification
    ) -> bool:
        """
        Submit scaling justification to AI quorum.
        Returns True if approved, False if rejected.
        """
        
        # Verify requirements
        if not justification.meets_requirements():
            logger.warning("scaling_justification_insufficient", extra={
                "justification_id": justification.justification_id,
                "confidence": justification.statistical_confidence,
                "sustained": justification.sustained_signal
            })
            return False
        
        # Create approval context
        context = ApprovalContext(
            context_id=f"scaling_approval_{justification.justification_id}",
            context_type="scaling",
            subject=justification.to_dict(),
            risk_score=abs(justification.expected_cost_impact),
            quality_score=justification.expected_quality_impact,
            policy_compliance=True,
            evidence=justification.supporting_metrics,
            supporting_data=justification.to_dict()
        )
        
        # Get quorum decision
        decision = ai_quorum_evaluate(context)
        
        # Record scaling attempt
        db = get_db()
        with db.transaction() as conn:
            conn.execute("""
                INSERT INTO scaling_history (
                    scaling_id, target_type, family_id,
                    from_count, to_count, justification,
                    approved, approval_decision, executed, created_at
                ) VALUES (%s, %s, %s, %s, %s, %s, %s, %s, FALSE, NOW())
            """, (
                f"scaling_{uuid.uuid4().hex}",
                justification.target_type,
                justification.family_id,
                justification.current_count,
                justification.proposed_count,
                json.dumps(justification.to_dict()),
                decision.approved,
                json.dumps({
                    "decision_id": decision.decision_id,
                    "reasoning": decision.decision_reasoning,
                    "unanimous": decision.unanimous
                })
            ))
        
        logger.info("scaling_decision", extra={
            "justification_id": justification.justification_id,
            "approved": decision.approved,
            "current": justification.current_count,
            "proposed": justification.proposed_count
        })
        
        return decision.approved
    
    def execute_scaling(self, justification: ScalingJustification):
        """Execute approved scaling"""
        current = justification.current_count
        target = justification.proposed_count
        delta = target - current
        
        if delta > 0:
            # Scale up
            for _ in range(delta):
                # Create global supervisors
                _lifecycle_manager.create_supervisor(
                    layer=SupervisorLayer.GLOBAL
                )
        elif delta < 0:
            # Scale down
            supervisors_to_retire = self._select_supervisors_for_retirement(abs(delta))
            for supervisor_id in supervisors_to_retire:
                _lifecycle_manager.retire_supervisor(
                    supervisor_id,
                    "scaling_down"
                )
        
        # Mark as executed
        db = get_db()
        with db.transaction() as conn:
            conn.execute("""
                UPDATE scaling_history
                SET executed = TRUE, executed_at = NOW()
                WHERE target_type = 'supervisor'
                  AND from_count = %s AND to_count = %s
                  AND executed = FALSE
                ORDER BY created_at DESC
                LIMIT 1
            """, (current, target))
        
        logger.info("scaling_executed", extra={
            "from": current,
            "to": target,
            "delta": delta
        })
    
    def _select_supervisors_for_retirement(self, count: int) -> List[str]:
        """Select supervisors for retirement (least active)"""
        db = get_db()
        with db.connection() as conn:
            cursor = conn.execute("""
                SELECT supervisor_id FROM supervisors
                WHERE status = 'ACTIVE' AND layer = 3
                ORDER BY interventions_count ASC, created_at DESC
                LIMIT %s
            """, (count,))
            return [row[0] for row in cursor.fetchall()]
```

```python
class InterventionEngine:

    def execute_intervention(
        self,
        supervisor_id: str,
        intervention_type: InterventionType,
        target_family_id: Optional[str],
        target_task_id: Optional[str],
        target_stage: Optional[Stage],

        trigger_evidence: Dict
    ) -> SupervisorIntervention:

        intervention = SupervisorIntervention(
            intervention_id=f"intervention_{uuid.uuid4().hex}",
            supervisor_id=supervisor_id,
            intervention_type=intervention_type,
            target_family_id=target_family_id,

            target_stage=target_stage,
            trigger_reason=trigger_reason,
            trigger_evidence=trigger_evidence,
            action_taken={},
            expected_outcome={}
        )
        
        # Execute specific intervention type
        if intervention_type == InterventionType.STOP_TASK:
            self._stop_task(intervention)
        elif intervention_type == InterventionType.ROLLBACK_STAGE:
            self._rollback_stage(intervention)
        elif intervention_type == InterventionType.QUARANTINE_TASK:
            self._quarantine_task(intervention)
        elif intervention_type == InterventionType.THROTTLE_FAMILY:
            self._throttle_family(intervention)
        elif intervention_type == InterventionType.RETRY_FROM_STAGE:
            self._retry_from_stage(intervention)
        elif intervention_type == InterventionType.REASSIGN_TASK:
            self._reassign_task(intervention)
        elif intervention_type == InterventionType.SCALE_AGENTS:
            self._scale_agents(intervention)
        elif intervention_type == InterventionType.ADJUST_PRESSURE:
            self._adjust_pressure(intervention)
        elif intervention_type == InterventionType.EMERGENCY_HALT:
            self._emergency_halt(intervention)
        
        # Persist intervention
        self._persist_intervention(intervention)
        
        # Update supervisor stats
        self._update_supervisor_stats(supervisor_id)
        
        # Create decision trace
        self._emit_decision_trace(intervention)
        
        logger.info("intervention_executed", extra={
            "intervention_id": intervention.intervention_id,
            "type": intervention_type.value,
            "supervisor_id": supervisor_id
        })
        
        return intervention
    
    def _stop_task(self, intervention: SupervisorIntervention):

            })

        with db.transaction() as conn:

                UPDATE task_queue
                SET current_stage = %s,
                    status = 'PENDING',
                    claimed_by = NULL,
                    claimed_at = NULL,
                    claim_deadline = NULL,
                    retry_count = retry_count + 1
                WHERE task_id = %s
            """, (prev_stage.value, intervention.target_task_id))
        
        intervention.action_taken = {
            "action": "stage_rollback",
            "from_stage": current_stage.value,
            "to_stage": prev_stage.value
        }
        intervention.expected_outcome = {
            "task_stage": prev_stage.value,
            "task_status": "PENDING"
        }
```

### 6.2 QUEUE SATURATION DETECTOR

```python
class QueueSaturationDetector:
    """
    Detects queue saturation using depth and latency signals.
    Requires sustained signals to prevent transient spike reactions.
    """
    
    def __init__(self):
        self.sustained_window_seconds = 300  # 5 minutes
        self.measurement_interval_seconds = 30
        self.history: Dict[Tuple[str, Stage], List[QueueMetrics]] = {}
    
    def measure_queue_metrics(self, family_id: str, stage: Stage) -> QueueMetrics:
        """Measure current queue metrics"""
        db = get_db()
        
        with db.connection() as conn:
            # Get queue depth
            cursor = conn.execute("""
                SELECT COUNT(*) FROM task_queue
                WHERE family_id = %s AND current_stage = %s AND status = 'PENDING'
            """, (family_id, stage.value))
            depth = cursor.fetchone()[0]
            
            # Get latency percentiles
            cursor = conn.execute("""
                SELECT 
                    PERCENTILE_CONT(0.5) WITHIN GROUP (ORDER BY EXTRACT(EPOCH FROM (NOW() - created_at))) as p50,
                    PERCENTILE_CONT(0.95) WITHIN GROUP (ORDER BY EXTRACT(EPOCH FROM (NOW() - created_at))) as p95,
                    PERCENTILE_CONT(0.99) WITHIN GROUP (ORDER BY EXTRACT(EPOCH FROM (NOW() - created_at))) as p99
                FROM task_queue
                WHERE family_id = %s AND current_stage = %s AND status = 'PENDING'
                  AND created_at > NOW() - INTERVAL '1 hour'
            """, (family_id, stage.value))
            
            row = cursor.fetchone()
            if row and row[0]:
                p50, p95, p99 = float(row[0]), float(row[1]), float(row[2])
            else:
                p50, p95, p99 = 0.0, 0.0, 0.0
            
            # Get oldest item age
            cursor = conn.execute("""
                SELECT EXTRACT(EPOCH FROM (NOW() - MIN(created_at)))
                FROM task_queue
                WHERE family_id = %s AND current_stage = %s AND status = 'PENDING'
            """, (family_id, stage.value))
            
            oldest = cursor.fetchone()[0]
            oldest_age = float(oldest) if oldest else 0.0
        
        metrics = QueueMetrics(
            family_id=family_id,
            stage=stage,
            depth=depth,
            latency_p50=p50,
            latency_p95=p95,
            latency_p99=p99,
            oldest_item_age_seconds=oldest_age
        )
        
        # Store in history
        key = (family_id, stage)
        if key not in self.history:
            self.history[key] = []
        self.history[key].append(metrics)
        
        # Trim old history
        cutoff = datetime.now() - timedelta(seconds=self.sustained_window_seconds * 2)
        self.history[key] = [m for m in self.history[key] if m.timestamp > cutoff]
        
        return metrics
    
    def detect_saturation(self, family_id: str, stage: Stage) -> Tuple[bool, float, str]:
        """
        Detect saturation at stage.
        Returns: (is_saturated, saturation_level, reason)
        """
        # Get recent metrics
        metrics = self.measure_queue_metrics(family_id, stage)
        
        # Check if signal is sustained
        if not self._is_signal_sustained(family_id, stage):
            return False, 0.0, "transient_spike"
        
        # Get pressure profile
        profile = self._get_pressure_profile(family_id)
        
        # Compute saturation signals
        depth_saturation = self._compute_depth_saturation(metrics, profile)
        latency_saturation = self._compute_latency_saturation(metrics, profile)
        
        # Combined saturation (max of signals)
        saturation_level = max(depth_saturation, latency_saturation)
        
        is_saturated = saturation_level > 0.3  # Minimum threshold
        
        reason = []
        if depth_saturation > 0.3:
            reason.append(f"depth={metrics.depth}")
        if latency_saturation > 0.3:
            reason.append(f"p99={metrics.latency_p99:.0f}s")
        
        reason_str = "; ".join(reason) if reason else "none"
        
        return is_saturated, saturation_level, reason_str
    
    def _is_signal_sustained(self, family_id: str, stage: Stage) -> bool:
        """Check if saturation signal is sustained over window"""
        key = (family_id, stage)
        history = self.history.get(key, [])
        
        if len(history) < 5:  # Need at least 5 measurements
            return False
        
        # Get recent measurements
        cutoff = datetime.now() - timedelta(seconds=self.sustained_window_seconds)
        recent = [m for m in history if m.timestamp > cutoff]
        
        if len(recent) < 5:
            return False
        
        # Check if all recent measurements show saturation
        profile = self._get_pressure_profile(family_id)
        
        saturated_count = 0
        for m in recent:
            depth_sat = self._compute_depth_saturation(m, profile)
            latency_sat = self._compute_latency_saturation(m, profile)
            if max(depth_sat, latency_sat) > 0.3:
                saturated_count += 1
        
        # Require 80% of measurements to show saturation
        return saturated_count >= len(recent) * 0.8
    
    def _compute_depth_saturation(self, metrics: QueueMetrics, profile: PressureProfile) -> float:
        """Compute saturation level from queue depth"""
        if metrics.depth <= profile.target_queue_depth:
            return 0.0
        elif metrics.depth >= profile.max_queue_depth:
            return 1.0
        else:
            # Linear interpolation
            range_size = profile.max_queue_depth - profile.target_queue_depth
            excess = metrics.depth - profile.target_queue_depth
            return excess / range_size
    
    def _compute_latency_saturation(self, metrics: QueueMetrics, profile: PressureProfile) -> float:
        """Compute saturation level from latency"""
        if metrics.latency_p99 <= profile.target_p95_latency_seconds:
            return 0.0
        elif metrics.latency_p99 >= profile.max_p99_latency_seconds:
            return 1.0
        else:
            # Linear interpolation
            range_size = profile.max_p99_latency_seconds - profile.target_p95_latency_seconds
            excess = metrics.latency_p99 - profile.target_p95_latency_seconds
            return excess / range_size
    
    def _get_pressure_profile(self, family_id: str) -> PressureProfile:
        """Get pressure profile for family"""
        db = get_db()
        
        with db.connection() as conn:
            cursor = conn.execute("""
                SELECT aggressive_threshold, conservative_threshold,
                       hysteresis_delta, cooldown_minutes, learning_rate
                FROM pressure_profiles
                WHERE family_id = %s
            """, (family_id,))
            
            row = cursor.fetchone()
        
        if row:
            return PressureProfile(
                family_id=family_id,
                aggressive_threshold=float(row[0]),
                conservative_threshold=float(row[1]),
                hysteresis_delta=float(row[2]),
                cooldown_minutes=int(row[3]),
                learning_rate=float(row[4])
            )
        else:
            # Return default
            return PressureProfile(family_id=family_id)

# Global detector
```

### 6.3 MULTI-HOP BACKPRESSURE ENGINE

```python
class BackpressureEngine:
    """
    Multi-hop backpressure propagation with attenuation.
    Maintains per-family isolation.
    """
    
    def __init__(self):
        self.max_propagation_hops = 3
        self.current_backpressure: Dict[Tuple[str, Stage], BackpressureSignal] = {}
    
    def compute_backpressure(self, family_id: str) -> Dict[Stage, float]:
        """
        Compute backpressure levels for all stages in family.
        Returns dict of stage -> backpressure level (0-1).
        """
        detector = get_saturation_detector()
        backpressure = {}
        
        # Measure saturation at each stage
        stage_saturations = {}
        for stage in ALL_STAGES:
            is_sat, level, reason = detector.detect_saturation(family_id, stage)
            stage_saturations[stage] = level
            
            if is_sat:
                logger.info("stage_saturated", extra={
                    "family_id": family_id,
                    "stage": stage.value,
                    "level": level,
                    "reason": reason
                })
        
        # Compute backpressure for each stage (including upstream propagation)
        for i, stage in enumerate(ALL_STAGES):
            # Local saturation
            local_bp = stage_saturations[stage]
            
            # Propagate from downstream stages
            downstream_bp = 0.0
            for hop in range(1, self.max_propagation_hops + 1):
                downstream_idx = i + hop
                if downstream_idx < len(ALL_STAGES):
                    downstream_stage = ALL_STAGES[downstream_idx]
                    downstream_saturation = stage_saturations[downstream_stage]
                    
                    # Apply attenuation
                    attenuation = ATTENUATION_FACTORS.get(hop, 0.0)
                    attenuated_bp = downstream_saturation * attenuation
                    
                    downstream_bp = max(downstream_bp, attenuated_bp)
            
            # Combined backpressure (max of local and downstream)
            combined_bp = max(local_bp, downstream_bp)
            backpressure[stage] = combined_bp
            
            # Store signal
            if combined_bp > 0:
                signal = BackpressureSignal(
                    family_id=family_id,
                    stage=stage,
                    level=combined_bp,
                    source="computed",
                    timestamp=datetime.now()
                )
                self.current_backpressure[(family_id, stage)] = signal
        
        return backpressure
    
    def get_backpressure_level(self, family_id: str, stage: Stage) -> float:
        """Get current backpressure level for stage"""
        key = (family_id, stage)
        signal = self.current_backpressure.get(key)
        
        if signal:
            # Check if signal is still fresh (< 1 minute old)
            age = (datetime.now() - signal.timestamp).total_seconds()
            if age < 60:
                return signal.level
        
        # Recompute if stale
        backpressure = self.compute_backpressure(family_id)
        return backpressure.get(stage, 0.0)
    
    def propagate_backpressure_to_upstream(
        self,
        family_id: str,
        source_stage: Stage,
        level: float
    ):
        """
        Manually propagate backpressure signal upstream.
        Used for emergency situations.
        """
        stage_idx = ALL_STAGES.index(source_stage)
        
        for hop in range(1, self.max_propagation_hops + 1):
            upstream_idx = stage_idx - hop
            if upstream_idx >= 0:
                upstream_stage = ALL_STAGES[upstream_idx]
                attenuation = ATTENUATION_FACTORS.get(hop, 0.0)
                attenuated_level = level * attenuation
                
                signal = BackpressureSignal(
                    family_id=family_id,
                    stage=upstream_stage,
                    level=attenuated_level,
                    source="emergency_propagation",
                    timestamp=datetime.now()
                )
                
                self.current_backpressure[(family_id, upstream_stage)] = signal
                
                logger.warning("backpressure_propagated", extra={
                    "family_id": family_id,
                    "from_stage": source_stage.value,
                    "to_stage": upstream_stage.value,
                    "level": attenuated_level
                })
    
    def clear_backpressure(self, family_id: str):
        """Clear all backpressure signals for family"""
        keys_to_remove = [k for k in self.current_backpressure.keys() if k[0] == family_id]
        for key in keys_to_remove:
            del self.current_backpressure[key]
        
        logger.info("backpressure_cleared", extra={"family_id": family_id})

# Global engine
_backpressure_engine = BackpressureEngine()
```

```python
class PressureModeController:
    """
    Controls pressure mode transitions with hysteresis and cooldown.
    Prevents oscillation through asymmetric thresholds.
    """
    
    def __init__(self):
        self.mode_history: Dict[str, List[Tuple[datetime, PressureMode]]] = {}
    
    def get_current_mode(self, family_id: str) -> PressureMode:
        """Get current pressure mode for family"""
        db = get_db()
        
        with db.connection() as conn:
            cursor = conn.execute("""
                SELECT current_mode FROM pressure_state
                WHERE family_id = %s
            """, (family_id,))
            
            row = cursor.fetchone()
        
        if row:
            return PressureMode(row[0])
        else:
            return PressureMode.NORMAL
    
    def evaluate_mode_transition(self, family_id: str) -> Optional[PressureMode]:
        """
        Evaluate if mode transition is needed.
        Returns new mode if transition warranted, None otherwise.
        """
        # Check cooldown
        if self._is_in_cooldown(family_id):
            return None
        
        # Get current state
        current_mode = self.get_current_mode(family_id)
        profile = self._get_pressure_profile(family_id)
        
        # Compute pressure score
        pressure_score = self._compute_pressure_score(family_id)
        
        # Determine new mode with hysteresis
        new_mode = self._apply_hysteresis(
            current_mode,
            pressure_score,
            profile
        )
        
        if new_mode != current_mode:
            return new_mode
        
        return None
    
    def transition_mode(self, family_id: str, new_mode: PressureMode, reason: str):
        """Transition to new pressure mode"""
        old_mode = self.get_current_mode(family_id)
        
        if old_mode == new_mode:
            return
        
        db = get_db()
        profile = self._get_pressure_profile(family_id)
        
        with db.transaction() as conn:
            # Update mode
            conn.execute("""
                UPDATE pressure_state
                SET current_mode = %s,
                    last_mode_change = NOW(),
                    mode_change_count = mode_change_count + 1,
                    cooldown_until = NOW() + INTERVAL '%s minutes',
                    updated_at = NOW()
                WHERE family_id = %s
            """, (new_mode.value, profile.cooldown_minutes, family_id))
            
            # Create intervention record
            intervention = get_intervention_engine().execute_intervention(
                supervisor_id=self._get_local_supervisor(family_id),
                intervention_type=InterventionType.ADJUST_PRESSURE,
                target_family_id=family_id,
                target_task_id=None,
                target_stage=None,
                trigger_reason=reason,
                trigger_evidence={
                    "old_mode": old_mode.value,
                    "new_mode": new_mode.value,
                    "pressure_score": self._compute_pressure_score(family_id)
                }
            )
        
        # Record in history
        if family_id not in self.mode_history:
            self.mode_history[family_id] = []
        self.mode_history[family_id].append((datetime.now(), new_mode))
        
        logger.info("pressure_mode_transition", extra={
            "family_id": family_id,
            "old_mode": old_mode.value,
            "new_mode": new_mode.value,
            "reason": reason
        })
    
    def _is_in_cooldown(self, family_id: str) -> bool:
        """Check if family is in mode change cooldown"""
        db = get_db()
        
        with db.connection() as conn:
            cursor = conn.execute("""
                SELECT cooldown_until FROM pressure_state
                WHERE family_id = %s
            """, (family_id,))
            
            row = cursor.fetchone()
        
        if row and row[0]:
            return datetime.now() < row[0]
        
        return False
    
    def _compute_pressure_score(self, family_id: str) -> float:
        """Compute composite pressure score (0-1)"""
        # Get backpressure levels
        bp_engine = get_backpressure_engine()
        backpressure = bp_engine.compute_backpressure(family_id)
        
        # Average backpressure across critical stages
        critical_stages = [Stage.EXECUTE, Stage.QA, Stage.DELIVER]
        critical_bp = [backpressure.get(s, 0.0) for s in critical_stages]
        avg_bp = sum(critical_bp) / len(critical_bp) if critical_bp else 0.0
        
        # Get quality metrics
        quality_score = self._get_recent_quality_score(family_id)
        quality_pressure = 1.0 - quality_score
        
        # Get failure rate
        failure_rate = self._get_failure_rate(family_id)
        
        # Composite score
        pressure = (
            0.4 * avg_bp +
            0.3 * quality_pressure +
            0.3 * failure_rate
        )
        
        return min(1.0, max(0.0, pressure))
    
    def _apply_hysteresis(
        self,
        current_mode: PressureMode,
        pressure_score: float,
        profile: PressureProfile
    ) -> PressureMode:
        """Apply hysteresis to prevent oscillation"""
        delta = profile.hysteresis_delta
        
        if current_mode == PressureMode.NORMAL:
            if pressure_score > profile.aggressive_threshold:
                return PressureMode.SOFT
        
        elif current_mode == PressureMode.SOFT:
            if pressure_score > profile.conservative_threshold + delta:
                return PressureMode.STEALTH
            elif pressure_score < profile.aggressive_threshold - delta:
                return PressureMode.NORMAL
        
        elif current_mode == PressureMode.STEALTH:
            if pressure_score > profile.critical_threshold:
                return PressureMode.HOLD
            elif pressure_score < profile.conservative_threshold - delta:
                return PressureMode.SOFT
        
        elif current_mode == PressureMode.HOLD:
            if pressure_score < profile.critical_threshold - delta:
                return PressureMode.STEALTH
        
        return current_mode
    
    def _get_pressure_profile(self, family_id: str) -> PressureProfile:
        """Get pressure profile for family"""
        return get_saturation_detector()._get_pressure_profile(family_id)
    
    def _get_recent_quality_score(self, family_id: str) -> float:
        """Get recent average quality score"""
        db = get_db()
        
        with db.connection() as conn:
            cursor = conn.execute("""
                SELECT AVG(final_quality_score)
                FROM quality_certificates
                WHERE family_id = %s
                  AND issued_at > NOW() - INTERVAL '1 hour'
            """, (family_id,))
            
            row = cursor.fetchone()
        
        if row and row[0]:
            return float(row[0])
        
        return 1.0  # Assume good if no data
    
    def _get_failure_rate(self, family_id: str) -> float:
        """Get recent failure rate"""
        db = get_db()
        
        with db.connection() as conn:
            cursor = conn.execute("""
                SELECT 
                    COUNT(CASE WHEN status = 'FAILED' THEN 1 END)::float /
                    NULLIF(COUNT(*)::float, 0)
                FROM task_queue
                WHERE family_id = %s
                  AND updated_at > NOW() - INTERVAL '1 hour'
            """, (family_id,))
            
            row = cursor.fetchone()
        
        if row and row[0]:
            return float(row[0])
        
        return 0.0
    
    def _get_local_supervisor(self, family_id: str) -> str:
        """Get local supervisor for family"""
        db = get_db()
        
        with db.connection() as conn:
            cursor = conn.execute("""
                SELECT supervisor_id FROM supervisors
                WHERE family_id = %s AND layer = 1 AND status = 'ACTIVE'
                LIMIT 1
            """, (family_id,))
            
            row = cursor.fetchone()
        
        if row:
            return row[0]
        
        # Create if missing
        supervisor = get_lifecycle_manager().create_supervisor(
            layer=SupervisorLayer.LOCAL,
            family_id=family_id
        )
        return supervisor.supervisor_id
```

### 6.5 THROTTLE EXECUTION ENGINE

```python
class ThrottleEngine:
    """
    Executes throttling actions based on pressure mode.
    Provides safe resume logic with verification.
    """
    
    def __init__(self):
        self.throttle_states: Dict[Tuple[str, Stage], ThrottleState] = {}
    
    def get_throttle_state(self, family_id: str, stage: Stage) -> ThrottleState:
        """Get current throttle state for stage"""
        key = (family_id, stage)
        return self.throttle_states.get(key, ThrottleState.OPEN)
    
    def apply_throttle(
        self,
        family_id: str,
        stage: Stage,
        mode: PressureMode
    ):
        """Apply throttle based on pressure mode"""
        key = (family_id, stage)
        
        # Determine throttle state from mode
        if mode == PressureMode.NORMAL:
            new_state = ThrottleState.OPEN
        elif mode == PressureMode.SOFT:
            new_state = ThrottleState.THROTTLED
        elif mode == PressureMode.STEALTH:
            new_state = ThrottleState.PAUSED
        elif mode == PressureMode.HOLD:
            new_state = ThrottleState.CLOSED
        else:
            new_state = ThrottleState.OPEN
        
        old_state = self.get_throttle_state(family_id, stage)
        
        if old_state != new_state:
            self.throttle_states[key] = new_state
            
            logger.info("throttle_state_changed", extra={
                "family_id": family_id,
                "stage": stage.value,
                "old_state": old_state.value,
                "new_state": new_state.value,
                "mode": mode.value
            })
    
    def should_process_stage(
        self,
        family_id: str,
        stage: Stage
    ) -> bool:
        """Check if stage should process items (respects throttle)"""
        state = self.get_throttle_state(family_id, stage)
        
        if state == ThrottleState.CLOSED:
            return False
        elif state == ThrottleState.PAUSED:
            # Allow minimal processing
            return self._allow_minimal_processing(family_id, stage)
        elif state == ThrottleState.THROTTLED:
            # Allow reduced processing
            return self._allow_throttled_processing(family_id, stage)
        else:
            return True
    
    def get_processing_capacity(
        self,
        family_id: str,
        stage: Stage,
        base_capacity: int
    ) -> int:
        """Get adjusted processing capacity based on throttle state"""
        state = self.get_throttle_state(family_id, stage)
        
        if state == ThrottleState.OPEN:
            return base_capacity
        elif state == ThrottleState.THROTTLED:
            return max(1, int(base_capacity * 0.5))
        elif state == ThrottleState.PAUSED:
            return max(1, int(base_capacity * 0.1))
        elif state == ThrottleState.CLOSED:
            return 0
        
        return base_capacity
    
    def verify_resume_conditions(self, family_id: str) -> Tuple[bool, str]:
        """
        Verify conditions for resuming from throttled state.
        Returns (can_resume, reason).
        """
        # Check backpressure levels
        bp_engine = get_backpressure_engine()
        backpressure = bp_engine.compute_backpressure(family_id)
        
        max_bp = max(backpressure.values()) if backpressure else 0.0
        
        if max_bp > 0.5:
            return False, f"backpressure_too_high: {max_bp:.2f}"
        
        # Check queue depths
        detector = get_saturation_detector()
        for stage in [Stage.EXECUTE, Stage.QA, Stage.DELIVER]:
            metrics = detector.measure_queue_metrics(family_id, stage)
            if metrics.depth > 1000:
                return False, f"queue_depth_high: {stage.value}={metrics.depth}"
        
        # Check quality
        quality = get_mode_controller()._get_recent_quality_score(family_id)
        if quality < MIN_QUALITY_SCORE:
            return False, f"quality_degraded: {quality:.4f}"
        
        return True, "conditions_met"
    
    def resume_processing(self, family_id: str):
        """Resume processing with verification"""
        can_resume, reason = self.verify_resume_conditions(family_id)
        
        if not can_resume:
            logger.warning("resume_denied", extra={
                "family_id": family_id,
                "reason": reason
            })
            return False
        
        # Clear throttles
        keys_to_remove = [k for k in self.throttle_states.keys() if k[0] == family_id]
        for key in keys_to_remove:
            self.throttle_states[key] = ThrottleState.OPEN
        
        # Reset pressure mode to NORMAL
        get_mode_controller().transition_mode(
            family_id,
            PressureMode.NORMAL,
            f"resume_verified: {reason}"
        )
        
        logger.info("processing_resumed", extra={
            "family_id": family_id,
            "reason": reason
        })
        
        return True
    
    def _allow_minimal_processing(self, family_id: str, stage: Stage) -> bool:
        """Allow minimal processing in PAUSED state"""
        # Allow 1 item per minute
        import random
        return random.random() < 0.1
    
    def _allow_throttled_processing(self, family_id: str, stage: Stage) -> bool:
        """Allow reduced processing in THROTTLED state"""
        # Allow 50% of normal processing
        import random
        return random.random() < 0.5

# Global throttle engine
_throttle_engine = ThrottleEngine()

def get_throttle_engine() -> ThrottleEngine:
    return _throttle_engine
```

```python
class PressureProfileLearner:
    """
    Adapts pressure profiles based on observed outcomes.
    Uses learning_rate to control adaptation speed.
    """
    
    def adapt_profile_from_outcomes(self, family_id: str):
        """Adapt profile based on recent outcomes"""
        profile = get_saturation_detector()._get_pressure_profile(family_id)
        
        # Get recent quality
        recent_quality = get_mode_controller()._get_recent_quality_score(family_id)
        
        # Get recent throughput
        recent_throughput = self._get_recent_throughput(family_id)
        
        # Adaptation logic
        if recent_quality < MIN_QUALITY_SCORE:
            # Quality suffered - become more conservative
            profile.conservative_threshold -= profile.learning_rate * 0.1
            profile.conservative_threshold = max(0.5, profile.conservative_threshold)
            
            profile.aggressive_threshold -= profile.learning_rate * 0.1
            profile.aggressive_threshold = max(0.2, profile.aggressive_threshold)
            
            logger.info("profile_adapted_conservative", extra={
                "family_id": family_id,
                "quality": recent_quality
            })
        
        elif recent_quality >= 0.99995 and recent_throughput > 0:
            # Quality excellent and throughput positive - can be more aggressive
            profile.aggressive_threshold += profile.learning_rate * 0.05
            profile.aggressive_threshold = min(0.5, profile.aggressive_threshold)
            
            logger.info("profile_adapted_aggressive", extra={
                "family_id": family_id,
                "quality": recent_quality,
                "throughput": recent_throughput
            })
        
        # Ensure thresholds remain ordered
        if profile.aggressive_threshold >= profile.conservative_threshold:
            profile.aggressive_threshold = profile.conservative_threshold - 0.1
        
        # Persist updated profile
        self._persist_profile(profile)
    
    def _get_recent_throughput(self, family_id: str) -> float:
        """Get recent throughput (tasks/hour)"""
        db = get_db()
        
        with db.connection() as conn:
            cursor = conn.execute("""
                SELECT COUNT(*)
                FROM task_queue
                WHERE family_id = %s
                  AND status = 'COMPLETED'
                  AND completed_at > NOW() - INTERVAL '1 hour'
            """, (family_id,))
            
            count = cursor.fetchone()[0]
        
        return float(count)
    
    def _persist_profile(self, profile: PressureProfile):
        """Persist updated profile to database"""
        db = get_db()
        
        with db.transaction() as conn:
            conn.execute("""
                UPDATE pressure_profiles
                SET aggressive_threshold = %s,
                    conservative_threshold = %s,
                    updated_at = NOW()
                WHERE family_id = %s
            """, (
                profile.aggressive_threshold,
                profile.conservative_threshold,
                profile.family_id
            ))
```

### 6.7 CONTINUOUS BACKPRESSURE MONITORING

```python
def run_backpressure_monitoring_loop():
    """
    Continuous backpressure monitoring and mode adjustment.
    Runs as daemon process.
    """
    bp_engine = get_backpressure_engine()
    mode_controller = get_mode_controller()
    throttle_engine = get_throttle_engine()
    
    cycle_count = 0
    
    while True:
        try:
            cycle_count += 1
            
            # Get active families
            db = get_db()
            with db.connection() as conn:
                cursor = conn.execute("""
                    SELECT family_id FROM families WHERE status = 'ACTIVE'
                """)
                families = [row[0] for row in cursor.fetchall()]
            
            for family_id in families:
                try:
                    # Compute backpressure
                    backpressure = bp_engine.compute_backpressure(family_id)
                    
                    # Evaluate mode transition
                    new_mode = mode_controller.evaluate_mode_transition(family_id)
                    
                    if new_mode:
                        current_mode = mode_controller.get_current_mode(family_id)
                        mode_controller.transition_mode(
                            family_id,
                            new_mode,
                            f"backpressure_triggered: max_bp={max(backpressure.values()):.2f}"
                        )
                        
                        # Apply throttles to all stages
                        for stage in ALL_STAGES:
                            throttle_engine.apply_throttle(family_id, stage, new_mode)
                    
                    # Emit metrics
                    for stage, level in backpressure.items():
                        record_metric(
                            "backpressure.level",
                            level,
                            {"family_id": family_id, "stage": stage.value}
                        )
                    
                    # Check for resume conditions if throttled
                    current_mode = mode_controller.get_current_mode(family_id)
                    if current_mode in [PressureMode.SOFT, PressureMode.STEALTH, PressureMode.HOLD]:
                        can_resume, _ = throttle_engine.verify_resume_conditions(family_id)
                        if can_resume and current_mode != PressureMode.HOLD:
                            throttle_engine.resume_processing(family_id)
                
                except Exception as e:
                    logger.error("backpressure_monitoring_family_error", extra={
                        "family_id": family_id,
                        "error": str(e)
                    })
            
            # Adaptive learning (every 10 cycles)
            if cycle_count % 10 == 0:
                run_adaptive_learning_cycle()
            
            # Sleep between cycles
            time.sleep(30)
            
        except Exception as e:
            logger.error("backpressure_monitoring_error", extra={"error": str(e)})
            time.sleep(60)
```

### 6.9 FAILURE SAFETY

```python
class BackpressureFailsafe:
    """
    Ensures backpressure system fails safely.
    If backpressure engine fails, default to CLOSED (safe mode).
    """
    
    @staticmethod
    def safe_get_backpressure_level(family_id: str, stage: Stage) -> float:
        """Get backpressure level with failsafe"""
        try:
            return get_backpressure_engine().get_backpressure_level(family_id, stage)
        except Exception as e:
            logger.critical("backpressure_engine_failed", extra={
                "family_id": family_id,
                "stage": stage.value,
                "error": str(e)
            })
            # Fail closed - return maximum backpressure
            return 1.0
    
    @staticmethod
    def safe_should_process(family_id: str, stage: Stage) -> bool:
        """Check if should process with failsafe"""
        try:
            return get_throttle_engine().should_process_stage(family_id, stage)
        except Exception as e:
            logger.critical("throttle_engine_failed", extra={
                "family_id": family_id,
                "stage": stage.value,
                "error": str(e)
            })
            # Fail closed - do not process
            return False
    
    @staticmethod
    def safe_get_capacity(family_id: str, stage: Stage, base: int) -> int:
        """Get capacity with failsafe"""
        try:
            return get_throttle_engine().get_processing_capacity(family_id, stage, base)
        except Exception as e:
            logger.critical("capacity_calculation_failed", extra={
                "family_id": family_id,
                "stage": stage.value,
                "error": str(e)
            })
            # Fail closed - return minimal capacity
```

```python
"""
STAGE PROCESSOR FRAMEWORK
Provides strict contract enforcement and consistency validation.
All stage processors inherit from this base.
"""

from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Optional, Dict, Any, List, Tuple
from datetime import datetime, timedelta
import uuid
import json
import logging

logger = logging.getLogger(__name__)

@dataclass
class StageProcessorInput:
    """Strict input contract for stage processors"""
    task_id: str
    family_id: str
    current_stage: Stage
    opportunity_data: Dict[str, Any]
    accumulated_stage_data: Dict[Stage, Dict[str, Any]]
    retry_count: int
    trace_id: str
    
    def validate(self):
        """Validate input contract"""
        if not self.task_id:
            raise ValueError("task_id required")
        if not self.family_id:
            raise ValueError("family_id required")
        if not isinstance(self.current_stage, Stage):
            raise ValueError("current_stage must be Stage enum")
        if not isinstance(self.accumulated_stage_data, dict):
            raise ValueError("accumulated_stage_data must be dict")

@dataclass
class StageProcessorOutput:
    """Strict output contract for stage processors"""
    success: bool
    next_stage: Optional[Stage]
    stage_data: Dict[str, Any]
    evidence_bundle: Optional[EvidenceBundle]
    decision_nodes: List[DecisionNode]
    failure_reason: Optional[str] = None
    failure_pattern: Optional[FailurePattern] = None
    
    def validate(self):
        """Validate output contract"""
        if self.success and not self.next_stage:
            if self.next_stage != Stage.POSTMORTEM:
                raise ValueError("Successful processing must provide next_stage")
        if not self.success and not self.failure_reason:
            raise ValueError("Failed processing must provide failure_reason")
        if not isinstance(self.stage_data, dict):
            raise ValueError("stage_data must be dict")
        if not isinstance(self.decision_nodes, list):
            raise ValueError("decision_nodes must be list")

class StageProcessor(ABC):
    """
    Base class for all stage processors.
    Enforces contracts, validates transitions, emits traces.
    """
    
    def __init__(self, stage: Stage):
        self.stage = stage
        self.processor_id = f"processor_{stage.value}_{uuid.uuid4().hex[:8]}"
    
    @abstractmethod
    def _process_impl(self, input_data: StageProcessorInput) -> StageProcessorOutput:
        """
        Implementation-specific processing logic.
        Subclasses must implement this.
        """
        pass
    
    def process(self, task_id: str) -> StageProcessorOutput:
        """
        Main entry point for processing.
        Handles claim, validation, execution, and persistence.
        """
        # Claim task
        claim_success, input_data = self._claim_task(task_id)
        
        if not claim_success:
            return StageProcessorOutput(
                success=False,
                next_stage=None,
                stage_data={},
                evidence_bundle=None,
                decision_nodes=[],
                failure_reason="claim_failed"
            )
        
        try:
            # Validate input
            input_data.validate()
            
            # Validate transition is legal
            if not self._validate_transition(input_data.current_stage):
                return self._handle_invalid_transition(input_data)
            
            # Check idempotency
            if self._is_already_processed(input_data):
                return self._get_cached_output(input_data)
            
            # Execute processing
            output = self._process_impl(input_data)
            
            # Validate output
            output.validate()
            
            # Validate next stage transition
            if output.success and output.next_stage:
                if not validate_transition(self.stage, output.next_stage):
                    raise ValueError(f"Invalid transition: {self.stage.value} -> {output.next_stage.value}")
            
            # Persist results
            self._persist_output(input_data, output)
            
            # Emit metrics
            self._emit_metrics(input_data, output)
            
            # Release claim
            self._release_claim(task_id, output.success)
            
            return output
            
        except Exception as e:
            logger.error("stage_processor_error", extra={
                "stage": self.stage.value,
                "task_id": task_id,
                "error": str(e)
            })
            
            # Create failure pattern
            failure_pattern = self._create_failure_pattern(input_data, e)
            
            # Release claim
            self._release_claim(task_id, False)
            
            return StageProcessorOutput(
                success=False,
                next_stage=Stage.POSTMORTEM,
                stage_data={},
                evidence_bundle=None,
                decision_nodes=[],
                failure_reason=str(e),
                failure_pattern=failure_pattern
            )
    
    def _claim_task(self, task_id: str) -> Tuple[bool, Optional[StageProcessorInput]]:
        """Claim task for processing"""
        db = get_db()
        
        claim_deadline = datetime.now() + timedelta(minutes=30)
        
        try:
            with db.transaction() as conn:
                # Atomic claim
                cursor = conn.execute("""
                    UPDATE task_queue
                    SET status = 'CLAIMED',
                        claimed_by = %s,
                        claimed_at = NOW(),
                        claim_deadline = %s
                    WHERE task_id = %s
                      AND status = 'PENDING'
                      AND current_stage = %s
                    RETURNING task_id, family_id, current_stage, 
                              opportunity_data, stage_data, retry_count
                """, (self.processor_id, claim_deadline, task_id, self.stage.value))
                
                row = cursor.fetchone()
                
                if not row:
                    return False, None
                
                # Build input
                accumulated_data = {}
                stage_data_json = json.loads(row[4]) if row[4] else {}
                
                for stage in ALL_STAGES:
                    stage_key = f"{stage.value}_data"
                    if stage_key in stage_data_json:
                        accumulated_data[stage] = stage_data_json[stage_key]
                
                input_data = StageProcessorInput(
                    task_id=row[0],
                    family_id=row[1],
                    current_stage=Stage(row[2]),
                    opportunity_data=json.loads(row[3]),
                    accumulated_stage_data=accumulated_data,
                    retry_count=row[5],
                    trace_id=f"trace_{task_id}"
                )
                
                return True, input_data
                
        except Exception as e:
            logger.error("claim_task_failed", extra={
                "task_id": task_id,
                "stage": self.stage.value,
                "error": str(e)
            })
            return False, None
    
    def _release_claim(self, task_id: str, success: bool):
        """Release task claim"""
        db = get_db()
        
        with db.transaction() as conn:
            conn.execute("""
                UPDATE task_queue
                SET claimed_by = NULL,
                    claimed_at = NULL,
                    claim_deadline = NULL
                WHERE task_id = %s
            """, (task_id,))
    
    def _validate_transition(self, current_stage: Stage) -> bool:
        """Validate we're processing correct stage"""
        return current_stage == self.stage
    
    def _handle_invalid_transition(self, input_data: StageProcessorInput) -> StageProcessorOutput:
        """Handle invalid stage transition"""
        logger.error("invalid_stage_transition", extra={
            "task_id": input_data.task_id,
            "expected_stage": self.stage.value,
            "actual_stage": input_data.current_stage.value
        })
        
        # Quarantine task
        self._quarantine_task(input_data.task_id, "invalid_stage_transition")
        
        return StageProcessorOutput(
            success=False,
            next_stage=Stage.POSTMORTEM,
            stage_data={},
            evidence_bundle=None,
            decision_nodes=[],
            failure_reason="invalid_stage_transition"
        )
    
    def _quarantine_task(self, task_id: str, reason: str):
        """Quarantine task"""
        db = get_db()
        
        with db.transaction() as conn:
            conn.execute("""
                UPDATE task_queue
                SET status = 'QUARANTINED'
                WHERE task_id = %s
            """, (task_id,))
            
            conn.execute("""
                INSERT INTO quarantine_log (
                    quarantine_id, task_id, family_id,
                    trigger_reason, details, quarantined_at
                )
                SELECT %s, task_id, family_id, %s, '{}', NOW()
                FROM task_queue
                WHERE task_id = %s
            """, (f"quarantine_{uuid.uuid4().hex}", reason, task_id))
    
    def _is_already_processed(self, input_data: StageProcessorInput) -> bool:
        """Check if stage already processed (idempotency)"""
        stage_key = f"{self.stage.value}_data"
        return stage_key in input_data.accumulated_stage_data
    
    def _get_cached_output(self, input_data: StageProcessorInput) -> StageProcessorOutput:
        """Get cached output for idempotent replay"""
        stage_key = f"{self.stage.value}_data"
        cached_data = input_data.accumulated_stage_data.get(stage_key, {})
        
        return StageProcessorOutput(
            success=True,
            next_stage=self._determine_next_stage(cached_data),
            stage_data=cached_data,
            evidence_bundle=None,
            decision_nodes=[]
        )
    
    def _determine_next_stage(self, stage_data: Dict) -> Optional[Stage]:
        """Determine next stage from output"""
        # Default: next in sequence
        stages = ALL_STAGES
        current_idx = stages.index(self.stage)
        
        if current_idx < len(stages) - 1:
            return stages[current_idx + 1]
        
        return None
    
    def _persist_output(self, input_data: StageProcessorInput, output: StageProcessorOutput):
        """Persist processing output"""
        db = get_db()
        
        with db.transaction() as conn:
            # Update task
            if output.success:
                # Update stage data
                stage_data_update = {
                    f"{self.stage.value}_data": output.stage_data
                }
                
                conn.execute("""
                    UPDATE task_queue
                    SET current_stage = %s,
                        status = 'PENDING',
                        stage_data = stage_data || %s::jsonb,
                        updated_at = NOW()
                    WHERE task_id = %s
                """, (
                    output.next_stage.value if output.next_stage else self.stage.value,
                    json.dumps(stage_data_update),
                    input_data.task_id
                ))
            else:
                conn.execute("""
                    UPDATE task_queue
                    SET status = 'FAILED',
                        last_error = %s,
                        retry_count = retry_count + 1,
                        updated_at = NOW()
                    WHERE task_id = %s
                """, (output.failure_reason, input_data.task_id))
            
            # Persist decision nodes
            for node in output.decision_nodes:
                self._persist_decision_node(conn, input_data.trace_id, node)
            
            # Persist evidence bundle if present
            if output.evidence_bundle:
                self._persist_evidence_bundle(conn, output.evidence_bundle)
            
            # Persist failure pattern if present
            if output.failure_pattern:
                self._persist_failure_pattern(conn, output.failure_pattern)
    
    def _persist_decision_node(self, conn, trace_id: str, node: DecisionNode):
        """Persist decision node to trace"""
        conn.execute("""
            INSERT INTO decision_traces (
                trace_id, task_id, family_id, root_decision, nodes,
                final_outcome, total_confidence, trace_started_at
            ) VALUES (%s, '', '', '', %s, '{}', 1.0, NOW())
            ON CONFLICT (trace_id) DO UPDATE
            SET nodes = decision_traces.nodes || %s::jsonb
        """, (
            trace_id,
            json.dumps([node.__dict__]),
            json.dumps([node.__dict__])
        ))
    
    def _persist_evidence_bundle(self, conn, evidence: EvidenceBundle):
        """Persist evidence bundle"""
        # Evidence stored in QA certificate, handled in QA stage
        pass
    
    def _persist_failure_pattern(self, conn, pattern: FailurePattern):
        """Persist failure pattern"""
        conn.execute("""
            INSERT INTO failure_patterns (
                pattern_id, family_id, pattern_type, pattern_signature,
                first_seen, last_seen, occurrence_count,
                root_cause, affected_stages, severity
            ) VALUES (%s, %s, %s, %s, NOW(), NOW(), 1, %s, %s, %s)
            ON CONFLICT (pattern_id) DO UPDATE
            SET last_seen = NOW(),
                occurrence_count = failure_patterns.occurrence_count + 1
        """, (
            pattern.pattern_id,
            pattern.family_id,
            pattern.pattern_type,
            json.dumps(pattern.pattern_signature),
            pattern.root_cause,
            [s.value for s in pattern.affected_stages],
            pattern.severity
        ))
    
    def _create_failure_pattern(self, input_data: StageProcessorInput, error: Exception) -> FailurePattern:
        """Create failure pattern from error"""
        return FailurePattern(
            pattern_id=f"pattern_{uuid.uuid4().hex}",
            family_id=input_data.family_id,
            pattern_type="stage_processing_error",
            pattern_signature={
                "stage": self.stage.value,
                "error_type": type(error).__name__,
                "error_message": str(error)[:200]
            },
            first_seen=datetime.now(),
            last_seen=datetime.now(),
            occurrence_count=1,
            root_cause=str(error)[:500],
            affected_stages=[self.stage],
            severity="medium"
        )
    
    def _emit_metrics(self, input_data: StageProcessorInput, output: StageProcessorOutput):
        """Emit processing metrics"""
        record_metric("stage.processed", 1, {
            "family_id": input_data.family_id,
            "stage": self.stage.value,
            "success": output.success
        })
        
        if not output.success:
            record_metric("stage.failed", 1, {
                "family_id": input_data.family_id,
                "stage": self.stage.value,
                "reason": output.failure_reason
            })
```

```python
class HuntProcessor(StageProcessor):

            output=enriched,
            confidence=0.9,
            stage=self.stage,
            agent_id=self.processor_id,
            agent_type="hunt_processor"
        ))
        
        # Check for duplicates (idempotency at opportunity level)
        is_duplicate = self._check_duplicate_opportunity(enriched, input_data.family_id)
        
        if is_duplicate:
            return StageProcessorOutput(
                success=False,
                next_stage=Stage.POSTMORTEM,
                stage_data={"reason": "duplicate_opportunity"},
                evidence_bundle=None,
                decision_nodes=decision_nodes,
                failure_reason="duplicate_opportunity"
            )
        
        stage_data = {
            "enriched_opportunity": enriched,
            "hunt_timestamp": datetime.now().isoformat(),
            "hunter_id": self.processor_id,
            "source": enriched.get("source_platform", "unknown")
        }
        
        return StageProcessorOutput(
            success=True,
            next_stage=Stage.FILTER_L1,
            stage_data=stage_data,
            evidence_bundle=None,
            decision_nodes=decision_nodes
        )
    
    def _validate_opportunity_structure(self, opp_data: Dict) -> bool:
        """Validate opportunity has required fields"""
        required = ["title", "description", "estimated_value"]
        return all(field in opp_data for field in required)
    
    def _enrich_opportunity(self, opp_data: Dict) -> Dict:
        """Enrich opportunity with metadata"""
        enriched = opp_data.copy()
        
        # Extract skills
        enriched["extracted_skills"] = self._extract_skills(opp_data.get("description", ""))
        
        # Classify complexity
        enriched["complexity_score"] = self._assess_complexity(opp_data)
        
        # Estimate competition
        enriched["competition_estimate"] = self._estimate_competition(opp_data)
        
        return enriched
    
    def _extract_skills(self, text: str) -> List[str]:
        """Extract skills from text"""
        skill_keywords = [
            "python", "javascript", "react", "node", "django",
            "machine learning", "data analysis", "sql", "aws",
            "design", "writing", "seo", "marketing"
        ]
        
        text_lower = text.lower()
        found = []
        
        for skill in skill_keywords:
            if skill in text_lower:
                found.append(skill)
        
        return found
    
    def _assess_complexity(self, opp_data: Dict) -> float:
        """Assess complexity (0-1)"""
        score = 0.5
        
        value = opp_data.get("estimated_value", 0)
        if value > 1000:
            score += 0.2
        
        desc_len = len(opp_data.get("description", ""))
        if desc_len > 500:
            score += 0.1
        
        return min(1.0, score)
    
    def _estimate_competition(self, opp_data: Dict) -> float:
        """Estimate competition (0-1)"""
        # Simplified heuristic
        return 0.5
    
    def _check_duplicate_opportunity(self, opp_data: Dict, family_id: str) -> bool:
        """Check if opportunity already exists"""
        external_id = opp_data.get("external_id")

                WHERE family_id = %s
                  AND opportunity_data->>'external_id' = %s
                  AND status != 'FAILED'
                LIMIT 1
            """, (family_id, external_id))
            
            return cursor.fetchone() is not None
```

```python
class FilterL1Processor(StageProcessor):
    """
    FILTER_L1 stage: Basic filtering.
    Validates opportunity meets minimum criteria.
    """
    
    def __init__(self):
        super().__init__(Stage.FILTER_L1)
    
    def _process_impl(self, input_data: StageProcessorInput) -> StageProcessorOutput:
        """Filter L1: Basic validation"""
        
        decision_nodes = []
        hunt_data = input_data.accumulated_stage_data.get(Stage.HUNT, {})
        enriched_opp = hunt_data.get("enriched_opportunity", input_data.opportunity_data)
        
        # Check 1: Minimum value
        value = enriched_opp.get("estimated_value", 0)
        min_value = 20  # $20 minimum
        
        value_check = value >= min_value
        
        decision_nodes.append(DecisionNode(
            node_id=f"node_{uuid.uuid4().hex}",
            decision_type="value_check",
            inputs={"value": value, "min_value": min_value},
            logic_applied=f"Check value >= {min_value}",
            output=value_check,
            confidence=1.0,
            stage=self.stage,
            agent_id=self.processor_id,
            agent_type="filter_l1_processor"
        ))
        
        if not value_check:
            return StageProcessorOutput(
                success=False,
                next_stage=Stage.POSTMORTEM,
                stage_data={"rejection_reason": "value_too_low"},
                evidence_bundle=None,
                decision_nodes=decision_nodes,
                failure_reason="value_below_minimum"
            )
        
        # Check 2: Has description
        description = enriched_opp.get("description", "")
        desc_check = len(description) >= 20
        
        decision_nodes.append(DecisionNode(
            node_id=f"node_{uuid.uuid4().hex}",
            decision_type="description_check",
            inputs={"description_length": len(description)},
            logic_applied="Check description length >= 20",
            output=desc_check,
            confidence=1.0,
            stage=self.stage,
            agent_id=self.processor_id,
            agent_type="filter_l1_processor"
        ))
        
        if not desc_check:
            return StageProcessorOutput(
                success=False,
                next_stage=Stage.POSTMORTEM,
                stage_data={"rejection_reason": "description_insufficient"},
                evidence_bundle=None,
                decision_nodes=decision_nodes,
                failure_reason="description_too_short"
            )
        
        # Check 3: Not spam
        is_spam = self._detect_spam(enriched_opp)
        
        decision_nodes.append(DecisionNode(
            node_id=f"node_{uuid.uuid4().hex}",
            decision_type="spam_check",
            inputs={"title": enriched_opp.get("title", "")},
            logic_applied="Check for spam keywords",
            output=not is_spam,
            confidence=0.85,
            stage=self.stage,
            agent_id=self.processor_id,
            agent_type="filter_l1_processor"
        ))
        
        if is_spam:
            return StageProcessorOutput(
                success=False,
                next_stage=Stage.POSTMORTEM,
                stage_data={"rejection_reason": "spam_detected"},
                evidence_bundle=None,
                decision_nodes=decision_nodes,
                failure_reason="spam_detected"
            )
        
        stage_data = {
            "l1_passed": True,
            "checks_passed": ["value", "description", "spam"],
            "filter_timestamp": datetime.now().isoformat()
        }
        
        return StageProcessorOutput(
            success=True,
            next_stage=Stage.FILTER_L2,
            stage_data=stage_data,
            evidence_bundle=None,
            decision_nodes=decision_nodes
        )
    
    def _detect_spam(self, opp_data: Dict) -> bool:
        """Detect spam patterns"""
        spam_keywords = ["urgent", "asap", "cheap", "free", "click here"]
        
        title = opp_data.get("title", "").lower()
        spam_count = sum(1 for kw in spam_keywords if kw in title)
        
        return spam_count >= 2
```

```python
class FilterL2Processor(StageProcessor):
    """
    FILTER_L2 stage: Advanced filtering.
    Checks feasibility and fit.
    """
    
    def __init__(self):
        super().__init__(Stage.FILTER_L2)
    
    def _process_impl(self, input_data: StageProcessorInput) -> StageProcessorOutput:
        """Filter L2: Advanced validation"""
        
        decision_nodes = []
        hunt_data = input_data.accumulated_stage_data.get(Stage.HUNT, {})
        enriched_opp = hunt_data.get("enriched_opportunity", input_data.opportunity_data)

        required_skills = enriched_opp.get("extracted_skills", [])
        family_skills = self._get_family_skills(input_data.family_id)
        
        skill_overlap = set(required_skills).intersection(set(family_skills))
        has_skill_match = len(skill_overlap) > 0 or len(required_skills) == 0
        
        decision_nodes.append(DecisionNode(
            node_id=f"node_{uuid.uuid4().hex}",
            decision_type="skill_match",
            inputs={
                "required_skills": required_skills,
                "family_skills": family_skills
            },
            logic_applied="Check skill overlap",
            output=has_skill_match,
            confidence=0.8,
            stage=self.stage,
            agent_id=self.processor_id,
            agent_type="filter_l2_processor"
        ))
        
        if not has_skill_match:
            return StageProcessorOutput(
                success=False,
                next_stage=Stage.POSTMORTEM,
                stage_data={"rejection_reason": "no_skill_match"},
                evidence_bundle=None,
                decision_nodes=decision_nodes,
                failure_reason="skills_incompatible"
            )
        
        # Check 2: ROI acceptable
        value = enriched_opp.get("estimated_value", 0)
        complexity = enriched_opp.get("complexity_score", 0.5)
        estimated_hours = complexity * 40  # Simple heuristic
        roi = value / max(estimated_hours, 1)
        
        min_roi = 15  # $15/hour minimum
        roi_acceptable = roi >= min_roi
        
        decision_nodes.append(DecisionNode(
```

```python
class QAProcessor(StageProcessor):
    """
    QA stage: Quality assurance with evidence-based certification.
    Layer 1: Rule-based validation
    Layer 2: Statistical sampling
    """
    
    def __init__(self):
        super().__init__(Stage.QA)
    
    def _process_impl(self, input_data: StageProcessorInput) -> StageProcessorOutput:
        """QA: Validate quality and issue certificate"""
        
        decision_nodes = []
        execute_data = input_data.accumulated_stage_data.get(Stage.EXECUTE, {})
        deliverables = execute_data.get("deliverables", [])
        contract_data = input_data.accumulated_stage_data.get(Stage.CONTRACT, {})
        contract = contract_data.get("contract", {})
        
        # Collect evidence
        evidence = self._collect_evidence(deliverables, contract, input_data)
        
        decision_nodes.append(DecisionNode(
            node_id=f"node_{uuid.uuid4().hex}",
            decision_type="evidence_collection",
            inputs={"deliverable_count": len(deliverables)},
            logic_applied="Collect multi-dimensional quality evidence",
            output={"evidence_complete": evidence.is_complete()},
            confidence=0.95,
            stage=self.stage,
            agent_id=self.processor_id,
            agent_type="qa_processor"
        ))
        
        # Layer 1: Rule-based
        layer1_score = self._compute_layer1_score(evidence)
        
        decision_nodes.append(DecisionNode(
            node_id=f"node_{uuid.uuid4().hex}",
            decision_type="layer1_validation",
            inputs=asdict(evidence),
            logic_applied="Rule-based quality validation",
            output=layer1_score,
            confidence=1.0,
            stage=self.stage,
            agent_id=self.processor_id,
            agent_type="qa_processor"
        ))
        
        # Layer 2: Sampling
        layer2_score = self._compute_layer2_score(evidence, deliverables)
        
        decision_nodes.append(DecisionNode(
            node_id=f"node_{uuid.uuid4().hex}",
            decision_type="layer2_validation",
            inputs={"sample_size": len(deliverables)},
            logic_applied="Statistical sampling validation",
            output=layer2_score,
            confidence=0.9,
            stage=self.stage,
            agent_id=self.processor_id,
            agent_type="qa_processor"
        ))
        
        # Final score
        final_score = evidence.compute_quality_score()
        
        # Create certificate
        certificate = QACertificate(
            certificate_id=f"cert_{uuid.uuid4().hex}",
            task_id=input_data.task_id,
            family_id=input_data.family_id,
            stage=self.stage,
            evidence_bundle=evidence,
            layer1_score=layer1_score,
            layer2_score=layer2_score,
            final_quality_score=final_score,
            meets_threshold=evidence.meets_threshold(),
            issued_at=datetime.now(),
            issued_by=self.processor_id
        )
        
        # Persist certificate
        self._persist_certificate(certificate)
        
        if not certificate.meets_threshold:
            return StageProcessorOutput(
                success=False,
                next_stage=Stage.POSTMORTEM,
                stage_data={"reason": "quality_below_threshold", "score": final_score},
                evidence_bundle=evidence,
                decision_nodes=decision_nodes,
                failure_reason=f"quality_score_{final_score:.4f}_below_threshold"
            )
        
        stage_data = {
            "qa_passed": True,
            "certificate_id": certificate.certificate_id,
            "final_quality_score": final_score,
            "qa_timestamp": datetime.now().isoformat()
        }
        
        return StageProcessorOutput(
            success=True,
            next_stage=Stage.DELIVER,
            stage_data=stage_data,
            evidence_bundle=evidence,
            decision_nodes=decision_nodes
        )
    
    def _collect_evidence(
        self,
        deliverables: List[Dict],
        contract: Dict,
        input_data: StageProcessorInput
    ) -> EvidenceBundle:
        """Collect comprehensive evidence"""
        
        # Content quality signals
        content_completeness = 1.0 if len(deliverables) >= len(contract.get("deliverables", [])) else 0.8
        content_accuracy = self._assess_content_accuracy(deliverables)
        content_format_validity = self._assess_format_validity(deliverables)
        content_language_quality = 0.95  # Simplified
        content_depth = self._assess_content_depth(deliverables)
        
        # Compliance signals
        compliance_tos = True
        compliance_legal = True
        compliance_policy = True
        compliance_safety = True
        compliance_ethical = True
        
        # Consistency signals
        consistency_semantic = self._assess_semantic_consistency(input_data)
        consistency_data = self._assess_data_consistency(deliverables)
        consistency_contract = self._assess_contract_consistency(deliverables, contract)
        
        # Delivery signals
        delivery_usability = self._assess_usability(deliverables)
        delivery_satisfaction_proxy = 0.95
        delivery_presentation = 0.9
        
        return EvidenceBundle(
            evidence_id=f"evidence_{uuid.uuid4().hex}",
            task_id=input_data.task_id,
            family_id=input_data.family_id,
            stage=self.stage,
            content_completeness=content_completeness,
            content_accuracy=content_accuracy,
            content_format_validity=content_format_validity,
            content_language_quality=content_language_quality,
            content_depth=content_depth,
            compliance_tos=compliance_tos,
            compliance_legal=compliance_legal,
            compliance_policy=compliance_policy,
            compliance_safety=compliance_safety,
            compliance_ethical=compliance_ethical,
            consistency_semantic=consistency_semantic,
            consistency_data=consistency_data,
            consistency_contract=consistency_contract,
            delivery_usability=delivery_usability,
            delivery_satisfaction_proxy=delivery_satisfaction_proxy,
            delivery_presentation=delivery_presentation,
            collected_at=datetime.now(),
            source_artifacts={"deliverables": deliverables}
        )
    
    def _assess_content_accuracy(self, deliverables: List[Dict]) -> float:
        """Assess content accuracy"""
        # Simplified: check all deliverables completed
        completed = sum(1 for d in deliverables if d.get("status") == "completed")
        return completed / max(len(deliverables), 1)
    
    def _assess_format_validity(self, deliverables: List[Dict]) -> float:
        """Assess format validity"""
        # Simplified: check artifacts exist
        has_artifacts = sum(1 for d in deliverables if d.get("artifact_url"))
        return has_artifacts / max(len(deliverables), 1)
    
    def _assess_content_depth(self, deliverables: List[Dict]) -> float:
        """Assess content depth"""
        return 0.95
    
    def _assess_semantic_consistency(self, input_data: StageProcessorInput) -> float:
        """Assess semantic consistency across stages"""
        return 0.95
    
    def _assess_data_consistency(self, deliverables: List[Dict]) -> float:
        """Assess data consistency"""
        return 0.95
    
    def _assess_contract_consistency(self, deliverables: List[Dict], contract: Dict) -> float:
        """Assess consistency with contract"""
        expected = len(contract.get("deliverables", []))
        actual = len(deliverables)
        return min(1.0, actual / max(expected, 1))
    
    def _assess_usability(self, deliverables: List[Dict]) -> float:
        """Assess usability"""
        return 0.95
    
    def _compute_layer1_score(self, evidence: EvidenceBundle) -> float:
        """Layer 1: Rule-based scoring"""
        # All compliance must pass
        if not all([
            evidence.compliance_tos,
            evidence.compliance_legal,
            evidence.compliance_policy,
            evidence.compliance_safety,
            evidence.compliance_ethical
        ]):
            return 0.0
        
        # Average content signals
        content_avg = (
            evidence.content_completeness +
            evidence.content_accuracy +
            evidence.content_format_validity +
            evidence.content_language_quality +
            evidence.content_depth
        ) / 5
        
        return content_avg
    
    def _compute_layer2_score(self, evidence: EvidenceBundle, deliverables: List[Dict]) -> float:
        """Layer 2: Sampling-based scoring"""
        # Deterministic sampling
        import hashlib
        
        sample_size = min(3, len(deliverables))
        
        if sample_size == 0:
            return 0.0
        
        # Sample indices deterministically
        task_hash = hashlib.sha256(evidence.task_id.encode()).hexdigest()
        seed = int(task_hash[:8], 16)
        
        import random
        rng = random.Random(seed)
        sample_indices = rng.sample(range(len(deliverables)), sample_size)
        
        # Score sampled deliverables
        scores = []
        for idx in sample_indices:
            deliv = deliverables[idx]
            score = 1.0 if deliv.get("status") == "completed" else 0.5
            scores.append(score)
        
        return sum(scores) / len(scores)
    
    def _persist_certificate(self, certificate: QACertificate):
        """Persist QA certificate"""
        db = get_db()
        
        with db.transaction() as conn:
            conn.execute("""
                INSERT INTO quality_certificates (
                    certificate_id, task_id, family_id, stage,
                    evidence_bundle, layer1_score, layer2_score,
                    final_quality_score, meets_threshold, threshold_used,
                    issued_at, issued_by, certificate_version, validation_method
                ) VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s)
            """, (
                certificate.certificate_id,
                certificate.task_id,
                certificate.family_id,
                certificate.stage.value,
                json.dumps(asdict(certificate.evidence_bundle)),
                certificate.layer1_score,
                certificate.layer2_score,
                certificate.final_quality_score,
                certificate.meets_threshold,
                certificate.threshold_used,
                certificate.issued_at,
                certificate.issued_by,
                certificate.certificate_version,
                certificate.validation_method
            ))
            
            # Update task with certificate
            conn.execute("""
                UPDATE task_queue
                SET quality_certificate_id = %s,
                    final_quality_score = %s
                WHERE task_id = %s
            """, (
                certificate.certificate_id,
                certificate.final_quality_score,
                certificate.task_id
            ))
```

```python
        lessons = self._extract_lessons(input_data, outcome)
        
        decision_nodes.append(DecisionNode(
            node_id=f"node_{uuid.uuid4().hex}",
            decision_type="postmortem_analysis",
            inputs={"outcome": outcome, "quality_score": quality_score},
            logic_applied="Extract lessons and record outcome",
            output=lessons,
            confidence=1.0,
            stage=self.stage,
            agent_id=self.processor_id,
            agent_type="postmortem_processor"
        ))
        
        # Update task status
        final_status = "COMPLETED" if outcome == "success" else "FAILED"
        
        stage_data = {
            "outcome": outcome,
            "lessons": lessons,
            "final_status": final_status,
            "postmortem_timestamp": datetime.now().isoformat()
        }
        
        # Persist final status
        self._update_final_status(input_data.task_id, final_status)
        
        return StageProcessorOutput(
            success=True,
            next_stage=None,  # Terminal stage
            stage_data=stage_data,
            evidence_bundle=None,
            decision_nodes=decision_nodes
        )
    
    def _extract_lessons(self, input_data: StageProcessorInput, outcome: str) -> Dict:
        """Extract lessons learned"""
        lessons = {
            "outcome": outcome,
            "stages_completed": len(input_data.accumulated_stage_data),
            "retry_count": input_data.retry_count,
            "improvements": []
        }
        
        if outcome == "failure":
            lessons["improvements"].append("Review failure patterns")
            lessons["improvements"].append("Adjust filters if needed")
        
        return lessons
    
    def _update_final_status(self, task_id: str, status: str):
        """Update final task status"""
        db = get_db()
        
        with db.transaction() as conn:
            conn.execute("""
                UPDATE task_queue
                SET status = %s,
                    completed_at = NOW()
                WHERE task_id = %s
            """, (status, task_id))
```

```python
class SupervisorRegistry:
    """
    Manages supervisor lifecycle: create, activate, suspend, retire.
    Enforces MIN/MAX constraints.
    """
    
    def __init__(self):
        self.heartbeat_timeout_seconds = 300
        self.max_heartbeat_failures = 3
    
    def initialize_baseline_supervisors(self):
        """Initialize baseline supervisor count"""
        current_count = self.get_active_supervisor_count()
        
        if current_count >= BASELINE_SUPERVISORS:
            logger.info("baseline_already_met", extra={"count": current_count})
            return
        
        needed = BASELINE_SUPERVISORS - current_count
        
        logger.info("initializing_baseline_supervisors", extra={
            "current": current_count,
            "baseline": BASELINE_SUPERVISORS,
            "creating": needed
        })
        
        for i in range(needed):
            self.create_supervisor(
                layer=SupervisorLayer.GLOBAL,
                reason="baseline_initialization"
            )
    
    def create_supervisor(
        self,
        layer: SupervisorLayer,
        reason: str,
        family_id: Optional[str] = None,
        domain: Optional[str] = None
    ) -> SupervisorEntity:
        """Create new supervisor"""
        
        # Check MAX constraint
        if layer == SupervisorLayer.GLOBAL:
            current_count = self.get_active_supervisor_count()
            if current_count >= MAX_SUPERVISORS:
                raise ValueError(f"Cannot exceed MAX_SUPERVISORS={MAX_SUPERVISORS}")
        
        # Generate name
        if layer == SupervisorLayer.LOCAL:
            name = f"Local-{family_id}"
        elif layer == SupervisorLayer.DOMAIN:
            name = f"Domain-{domain}"
        else:
            name = f"Global-{uuid.uuid4().hex[:8]}"
        
        supervisor = SupervisorEntity(
            supervisor_id=f"supervisor_{uuid.uuid4().hex}",
            supervisor_name=name,
            layer=layer,
            state=SupervisorState.INITIALIZING,
            family_id=family_id,
            domain=domain
        )
        
        # Persist
        self._persist_supervisor(supervisor)
        
        # Activate immediately
        self.activate_supervisor(supervisor.supervisor_id)
        
        # Log audit event
        self._log_audit_event(
            event_type="supervisor_created",
            supervisor_id=supervisor.supervisor_id,
            details={
                "layer": layer.value,
                "reason": reason,
                "family_id": family_id,
                "domain": domain
            }

        logger.info("supervisor_created", extra={
            "supervisor_id": supervisor.supervisor_id,
            "layer": layer.value,
            "family_id": family_id,
            "domain": domain

        return supervisor

                    self.create_supervisor(
                        layer=SupervisorLayer(layer),
                        reason="failover_replacement",
                        family_id=family_id,
                        domain=domain
                    )
            else:
                # Increment failure count
                with db.transaction() as conn:
                    conn.execute("""
                        UPDATE supervisors
                        SET heartbeat_failures = %s
```

```python
@dataclass
class ScalingJustification:
    """Evidence-based scaling justification"""
    justification_id: str
    target: str  # "supervisors", "workers", "hunters"
    family_id: Optional[str]
    
    # Current state
    current_count: int

    expected_cost_delta: float

    def meets_requirements(self) -> bool:

        if self.statistical_confidence < SCALING_CONFIDENCE_THRESHOLD:
            return False
        if not self.sustained_signal:
            return False
        if self.signal_duration_hours < SCALING_SUSTAINED_SIGNAL_HOURS:
            return False
        return True
    
    def to_dict(self) -> Dict:
        return {
            "justification_id": self.justification_id,
            "target": self.target,
            "family_id": self.family_id,
            "current_count": self.current_count,
            "proposed_count": self.proposed_count,
            "delta": self.delta,
            "statistical_confidence": self.statistical_confidence,
            "data_points": self.data_points,
            "observation_window_hours": self.observation_window_hours,
            "sustained_signal": self.sustained_signal,
            "signal_duration_hours": self.signal_duration_hours,
            "queue_depth_avg": self.queue_depth_avg,
            "latency_p99_avg": self.latency_p99_avg,
            "failure_rate": self.failure_rate,
            "intervention_rate": self.intervention_rate,
            "expected_quality_delta": self.expected_quality_delta,
            "expected_throughput_delta": self.expected_throughput_delta,
            "expected_cost_delta": self.expected_cost_delta,
            "rollback_plan": self.rollback_plan
        }

class ScalingGovernanceEngine:
    """
    Manages scaling decisions with strict governance.
    Enforces MIN/MAX/BASELINE constraints.
    """
    
    def evaluate_supervisor_scaling_need(self) -> Optional[ScalingJustification]:
        """Evaluate if supervisor scaling is needed"""
        
        # Check cooldown
        if self._is_in_cooldown("supervisors"):
            logger.debug("scaling_in_cooldown", extra={"target": "supervisors"})
            return None
        
        # Get current count
        current_count = get_supervisor_registry().get_active_supervisor_count()
        
        # Gather metrics
        metrics = self._gather_supervisor_metrics()
        
        # Check if metrics meet quality bar
        if metrics["data_points"] < 1000:
            logger.debug("insufficient_data_points", extra={"points": metrics["data_points"]})
            return None
        
        # Compute scaling need
        scaling_decision = self._compute_supervisor_scaling_decision(current_count, metrics)
        
        if not scaling_decision["needs_scaling"]:
            return None
        
        proposed_count = scaling_decision["proposed_count"]
        
        # Enforce bounds
        if proposed_count < MIN_SUPERVISORS or proposed_count > MAX_SUPERVISORS:
            logger.warning("scaling_out_of_bounds", extra={
                "proposed": proposed_count,
                "min": MIN_SUPERVISORS,
                "max": MAX_SUPERVISORS
            })
            return None
        
        # Check sustained signal
        sustained = self._check_sustained_signal(metrics)
        
        if not sustained["is_sustained"]:
            logger.debug("signal_not_sustained", extra={
                "duration": sustained["duration_hours"]
            })
            return None
        
        # Build justification
        justification = ScalingJustification(
            justification_id=f"justification_{uuid.uuid4().hex}",
            target="supervisors",
            family_id=None,
            current_count=current_count,
            proposed_count=proposed_count,
            delta=proposed_count - current_count,
            statistical_confidence=metrics["confidence"],
            data_points=metrics["data_points"],
            observation_window_hours=metrics["window_hours"],
            sustained_signal=sustained["is_sustained"],
            signal_duration_hours=sustained["duration_hours"],
            queue_depth_avg=metrics["queue_depth_avg"],
            latency_p99_avg=metrics["latency_p99_avg"],
            failure_rate=metrics["failure_rate"],
            intervention_rate=metrics["intervention_rate"],
            expected_quality_delta=scaling_decision["quality_delta"],
            expected_throughput_delta=scaling_decision["throughput_delta"],
            expected_cost_delta=scaling_decision["cost_delta"],
            rollback_plan=self._create_rollback_plan(current_count, proposed_count)
        )
        
        return justification
    
    def submit_for_approval(self, justification: ScalingJustification) -> bool:
        """Submit scaling justification to AI quorum"""
        
        # Verify requirements
        if not justification.meets_requirements():
            logger.warning("justification_insufficient", extra={
                "justification_id": justification.justification_id,
                "confidence": justification.statistical_confidence,
                "sustained": justification.sustained_signal
            })
            return False
        
        # Create approval context
        context = ApprovalContext(
            context_id=f"scaling_{justification.justification_id}",
            context_type="scaling",
            subject=justification.to_dict(),
            risk_score=abs(justification.expected_cost_delta),
            quality_score=max(0.0, justification.expected_quality_delta),
            policy_compliance=True,
            evidence={
                "statistical_confidence": justification.statistical_confidence,
                "sustained_signal": justification.sustained_signal,
                "data_points": justification.data_points
            },
            supporting_data=justification.to_dict()
        )
        
        # Get AI quorum decision (3 judges)
        decision = ai_quorum_evaluate(context)
        
        # Persist scaling attempt
        self._persist_scaling_attempt(justification, decision)
        
        logger.info("scaling_decision", extra={
            "justification_id": justification.justification_id,
            "approved": decision.approved,
            "current": justification.current_count,
            "proposed": justification.proposed_count,
            "unanimous": decision.unanimous
        })
        
        return decision.approved
    
    def execute_scaling(self, justification: ScalingJustification):
        """Execute approved scaling"""
        
        delta = justification.delta
        
        if delta > 0:
            # Scale up
            for i in range(delta):
                get_supervisor_registry().create_supervisor(
                    layer=SupervisorLayer.GLOBAL,
                    reason=f"scaling_approved_{justification.justification_id}"
                )
        elif delta < 0:
            # Scale down
            supervisors_to_retire = self._select_supervisors_for_retirement(abs(delta))
            
            for supervisor_id in supervisors_to_retire:
                get_supervisor_registry().retire_supervisor(
                    supervisor_id,
                    f"scaling_down_{justification.justification_id}"
                )
        
        # Mark as executed
        self._mark_scaling_executed(justification.justification_id)
        
        logger.info("scaling_executed", extra={
            "justification_id": justification.justification_id,
            "from": justification.current_count,
            "to": justification.proposed_count,
            "delta": delta
        })
    
    def _is_in_cooldown(self, target: str) -> bool:
        """Check if in scaling cooldown"""
        db = get_db()
        
        with db.connection() as conn:
            cursor = conn.execute("""
                SELECT MAX(executed_at) FROM scaling_history
                WHERE target = %s AND executed = TRUE
            """, (target,))
            
            row = cursor.fetchone()
        
        if not row or not row[0]:
            return False
        
        last_scaling = row[0]
        hours_since = (datetime.now() - last_scaling).total_seconds() / 3600
        
        return hours_since < SCALING_COOLDOWN_HOURS
    
    def _gather_supervisor_metrics(self) -> Dict:
        """Gather metrics for supervisor scaling decision"""
        db = get_db()
        
        # Query metrics from last 48 hours
        window_start = datetime.now() - timedelta(hours=48)
        
        with db.connection() as conn:
            # Intervention rate
            cursor = conn.execute("""
                SELECT COUNT(*) FROM supervisor_interventions
                WHERE intervened_at > %s
            """, (window_start,))
            intervention_count = cursor.fetchone()[0]
            
            # Queue depths
            cursor = conn.execute("""
                SELECT AVG(
                    (SELECT COUNT(*) FROM task_queue
                     WHERE family_id = f.family_id AND status = 'PENDING')
                )
                FROM families f
                WHERE f.status = 'ACTIVE'
            """)
            queue_depth_avg = float(cursor.fetchone()[0] or 0)
            
            # Latency p99 (approximate from task ages)
            cursor = conn.execute("""
                SELECT PERCENTILE_CONT(0.99) WITHIN GROUP (
                    ORDER BY EXTRACT(EPOCH FROM (NOW() - created_at))
                )
                FROM task_queue
                WHERE status = 'PENDING' AND created_at > %s
            """, (window_start,))
            latency_p99 = float(cursor.fetchone()[0] or 0)
            
            # Failure rate
            cursor = conn.execute("""
                SELECT 
                    COUNT(CASE WHEN status = 'FAILED' THEN 1 END)::float /
                    NULLIF(COUNT(*)::float, 0)
                FROM task_queue
                WHERE updated_at > %s
            """, (window_start,))
            failure_rate = float(cursor.fetchone()[0] or 0)
        
        # Compute statistical confidence using bootstrap
        confidence = self._compute_bootstrap_confidence([intervention_count] * 100)
        
        return {
            "data_points": intervention_count + 100,
            "window_hours": 48,
            "confidence": confidence,
            "intervention_rate": intervention_count / 48.0,
            "queue_depth_avg": queue_depth_avg,
            "latency_p99_avg": latency_p99,
            "failure_rate": failure_rate
        }
    
    def _compute_supervisor_scaling_decision(self, current_count: int, metrics: Dict) -> Dict:
        """Compute scaling decision from metrics"""
        
        # Intervention rate per supervisor
        interventions_per_supervisor = metrics["intervention_rate"] / max(current_count, 1)
        
        # Thresholds
        overload_threshold = 5.0  # interventions per hour per supervisor
        underload_threshold = 1.0
        
        needs_scaling = False
        proposed_count = current_count
        
        if interventions_per_supervisor > overload_threshold:
            # Scale up
            scale_factor = min(1.5, interventions_per_supervisor / overload_threshold)
            proposed_count = min(MAX_SUPERVISORS, int(current_count * scale_factor))
            needs_scaling = True
        elif interventions_per_supervisor < underload_threshold and current_count > BASELINE_SUPERVISORS:
            # Scale down (but not below baseline)
            scale_factor = max(0.8, interventions_per_supervisor / underload_threshold)
            proposed_count = max(BASELINE_SUPERVISORS, int(current_count * scale_factor))
            needs_scaling = proposed_count != current_count
        
        # Compute expected impacts
        if needs_scaling:
            delta_ratio = (proposed_count - current_count) / current_count
            quality_delta = 0.03 * delta_ratio
            throughput_delta = 0.05 * delta_ratio
            cost_delta = delta_ratio
        else:
            quality_delta = 0.0
            throughput_delta = 0.0
            cost_delta = 0.0
        
        return {
            "needs_scaling": needs_scaling,
            "proposed_count": proposed_count,
            "quality_delta": quality_delta,
            "throughput_delta": throughput_delta,
            "cost_delta": cost_delta
        }
    
    def _check_sustained_signal(self, metrics: Dict) -> Dict:
        """Check if signal sustained over required window"""
        duration_hours = metrics["window_hours"]
        is_sustained = duration_hours >= SCALING_SUSTAINED_SIGNAL_HOURS
        
        return {
            "is_sustained": is_sustained,
            "duration_hours": duration_hours
        }
    
    def _compute_bootstrap_confidence(self, data: List[float]) -> float:
        """Compute bootstrap confidence"""
        if len(data) < 50:
            return 0.0
        
        import random
        
        n_bootstrap = 500
        means = []
        
        for _ in range(n_bootstrap):
            sample = random.choices(data, k=len(data))
            means.append(sum(sample) / len(sample))
        
        means.sort()
        lower = means[int(0.025 * n_bootstrap)]
        upper = means[int(0.975 * n_bootstrap)]
        mean_val = sum(data) / len(data)
        
        if mean_val == 0:
            return 1.0
        
        interval_width = (upper - lower) / mean_val
        confidence = max(0.0, 1.0 - interval_width)
        
        return confidence
    
    def _create_rollback_plan(self, current: int, proposed: int) -> Dict:
        """Create rollback plan"""
        return {

            "rollback_trigger": "quality_degradation or intervention_spike",
            "rollback_threshold": "quality_score < 0.9999 or intervention_rate > 2x baseline",
            "rollback_procedure": "retire newest supervisors first"
        }

                ) VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s, FALSE, NOW())
            """, (
                f"scaling_{uuid.uuid4().hex}",
                justification.target,
                justification.family_id,
                justification.current_count,
                justification.proposed_count,
                json.dumps(justification.to_dict()),
                justification.statistical_confidence,
                justification.sustained_signal,
                decision.approved,
                json.dumps({
                    "decision_id": decision.decision_id,
```

```python
@dataclass
class PressureProfile:
    """Per-family pressure configuration"""
    family_id: str
    
    # Mode thresholds (with learning)
    threshold_soft: float = 0.3
    threshold_stealth: float = 0.6
    threshold_hold: float = 0.9
    
    # Hysteresis
    hysteresis_delta: float = PRESSURE_HYSTERESIS_DELTA
    
    # Cooldown
    cooldown_minutes: int = PRESSURE_COOLDOWN_MINUTES
    
    # Learning
    learning_rate: float = 0.05
    
    # State
    current_mode: PressureMode = PressureMode.NORMAL
    last_mode_change: Optional[datetime] = None
    mode_change_count: int = 0
    cooldown_until: Optional[datetime] = None
    
    # Oscillation guard
    recent_transitions: List[Tuple[datetime, PressureMode]] = field(default_factory=list)
    
    def is_in_cooldown(self) -> bool:
        """Check if in cooldown period"""
        if not self.cooldown_until:
            return False
        return datetime.now() < self.cooldown_until
    
    def check_oscillation(self) -> bool:
        """Check if oscillating (3+ transitions in 1 hour)"""
        cutoff = datetime.now() - timedelta(hours=1)
        recent = [t for t in self.recent_transitions if t[0] > cutoff]
        return len(recent) >= 3

class PressureControlEngine:
    """
    Manages per-family pressure modes with hysteresis and oscillation prevention.
    """
    
    def __init__(self):
        self.profiles: Dict[str, PressureProfile] = {}
    
    def get_profile(self, family_id: str) -> PressureProfile:
        """Get or create pressure profile"""
        if family_id not in self.profiles:
            self.profiles[family_id] = self._load_profile(family_id)
        return self.profiles[family_id]
    
    def compute_pressure_score(self, family_id: str) -> float:
        """Compute current pressure score (0-1)"""
        
        # Get backpressure
        bp_engine = get_backpressure_engine()
        backpressure_levels = bp_engine.compute_family_backpressure(family_id)
        
        # Average across critical stages
        critical_stages = [Stage.EXECUTE, Stage.QA, Stage.DELIVER]
        bp_values = [backpressure_levels.get(s, 0.0) for s in critical_stages]
        avg_bp = sum(bp_values) / len(bp_values) if bp_values else 0.0
        
        # Get quality score
        quality = self._get_recent_quality_score(family_id)
        quality_pressure = 1.0 - quality
        
        # Get failure rate
        failure_rate = self._get_failure_rate(family_id)
        
        # Composite pressure score
        pressure = (
            0.4 * avg_bp +
            0.3 * quality_pressure +
            0.3 * failure_rate
        )
        
        return min(1.0, max(0.0, pressure))
    
    def evaluate_mode_transition(self, family_id: str) -> Optional[PressureMode]:
        """Evaluate if mode transition needed"""
        
        profile = self.get_profile(family_id)
        
        # Check cooldown
        if profile.is_in_cooldown():
            return None
        
        # Check oscillation
        if profile.check_oscillation():
            logger.warning("pressure_oscillation_detected", extra={"family_id": family_id})
            return None
        
        # Compute pressure
        pressure = self.compute_pressure_score(family_id)
        
        # Apply hysteresis
        new_mode = self._apply_hysteresis(profile.current_mode, pressure, profile)
        
        if new_mode != profile.current_mode:
            return new_mode
        
        return None
    
    def transition_mode(self, family_id: str, new_mode: PressureMode, reason: str):
        """Transition to new pressure mode"""
        
        profile = self.get_profile(family_id)
        old_mode = profile.current_mode
        
        if old_mode == new_mode:
            return
        
        # Update profile
        profile.current_mode = new_mode
        profile.last_mode_change = datetime.now()
        profile.mode_change_count += 1
        profile.cooldown_until = datetime.now() + timedelta(minutes=profile.cooldown_minutes)
        profile.recent_transitions.append((datetime.now(), new_mode))
        
        # Trim old transitions
        cutoff = datetime.now() - timedelta(hours=2)
        profile.recent_transitions = [t for t in profile.recent_transitions if t[0] > cutoff]
        
        # Persist
        self._persist_profile(profile)
        
        # Create intervention
        supervisor = get_supervisor_registry().get_local_supervisor(family_id)
        self._record_intervention(
            supervisor.supervisor_id,
            InterventionType.ADJUST_PRESSURE,
            family_id,
            reason,
            {"old_mode": old_mode.value, "new_mode": new_mode.value}
        )
        
        logger.info("pressure_mode_transition", extra={
            "family_id": family_id,
            "old_mode": old_mode.value,
            "new_mode": new_mode.value,
            "reason": reason
        })
    
    def adapt_thresholds(self, family_id: str):
        """Adapt thresholds based on outcomes"""
        
        profile = self.get_profile(family_id)
        
        # Get recent quality
        quality = self._get_recent_quality_score(family_id)
        
        if quality < MIN_QUALITY_SCORE:
            # Quality suffered - become more conservative
            profile.threshold_soft = max(0.2, profile.threshold_soft - profile.learning_rate * 0.1)
            profile.threshold_stealth = max(0.5, profile.threshold_stealth - profile.learning_rate * 0.1)
            
            logger.info("thresholds_adapted_conservative", extra={
                "family_id": family_id,
                "quality": quality
            })
        elif quality >= 0.99995:
            # Quality excellent - can be more aggressive
            profile.threshold_soft = min(0.4, profile.threshold_soft + profile.learning_rate * 0.05)
            profile.threshold_stealth = min(0.7, profile.threshold_stealth + profile.learning_rate * 0.05)
            
            logger.info("thresholds_adapted_aggressive", extra={
                "family_id": family_id,
                "quality": quality
            })
        
        # Ensure thresholds remain ordered
        if profile.threshold_soft >= profile.threshold_stealth:
            profile.threshold_soft = profile.threshold_stealth - 0.1
        if profile.threshold_stealth >= profile.threshold_hold:
            profile.threshold_stealth = profile.threshold_hold - 0.1
        
        # Persist
        self._persist_profile(profile)
    
    def _apply_hysteresis(
        self,
        current_mode: PressureMode,
        pressure: float,
        profile: PressureProfile
    ) -> PressureMode:
        """Apply hysteresis to prevent oscillation"""
        
        delta = profile.hysteresis_delta
        
        if current_mode == PressureMode.NORMAL:
            if pressure > profile.threshold_soft:
                return PressureMode.SOFT
        
        elif current_mode == PressureMode.SOFT:
            if pressure > profile.threshold_stealth + delta:
                return PressureMode.STEALTH
            elif pressure < profile.threshold_soft - delta:
                return PressureMode.NORMAL
        
        elif current_mode == PressureMode.STEALTH:
            if pressure > profile.threshold_hold:
                return PressureMode.HOLD
            elif pressure < profile.threshold_stealth - delta:
                return PressureMode.SOFT
        
        elif current_mode == PressureMode.HOLD:
            if pressure < profile.threshold_hold - delta:
                return PressureMode.STEALTH
        
        return current_mode
    
    def _load_profile(self, family_id: str) -> PressureProfile:
        """Load profile from DB"""
        db = get_db()
        
        with db.connection() as conn:
            cursor = conn.execute("""
                SELECT current_mode, threshold_soft, threshold_stealth, threshold_hold,
                       learning_rate, last_mode_change, mode_change_count, cooldown_until
                FROM pressure_profiles
                WHERE family_id = %s
            """, (family_id,))
            
            row = cursor.fetchone()
        
        if row:
            return PressureProfile(
                family_id=family_id,

                threshold_stealth=float(row[2]),
                threshold_hold=float(row[3]),
                learning_rate=float(row[4]),
                last_mode_change=row[5],
                mode_change_count=row[6],
                cooldown_until=row[7]
            )
        else:
            # Create default
            profile = PressureProfile(family_id=family_id)
            self._persist_profile(profile)
            return profile
    
    def _persist_profile(self, profile: PressureProfile):
        """Persist profile to DB"""
        db = get_db()
```

### 8.5 BACKPRESSURE ENGINE (MULTI-HOP)

```python
class BackpressureEngine:
    """
    Multi-hop backpressure with attenuation.
    Propagates upstream up to 3 hops.
    """
    
    def compute_family_backpressure(self, family_id: str) -> Dict[Stage, float]:
        """Compute backpressure for all stages in family"""
        
        # Measure saturation at each stage
        stage_saturations = {}
        
        for stage in ALL_STAGES:
            saturation = self._measure_stage_saturation(family_id, stage)
            stage_saturations[stage] = saturation
        
        # Compute backpressure with upstream propagation
        backpressure = {}
        
        for i, stage in enumerate(ALL_STAGES):
            # Local saturation
            local_bp = stage_saturations[stage]
            
            # Downstream propagation
            downstream_bp = 0.0
            
            for hop in range(1, BACKPRESSURE_MAX_HOPS + 1):
                downstream_idx = i + hop
                
                if downstream_idx < len(ALL_STAGES):
                    downstream_stage = ALL_STAGES[downstream_idx]
                    downstream_saturation = stage_saturations[downstream_stage]
                    
                    # Apply attenuation
                    attenuation = BACKPRESSURE_ATTENUATION.get(hop, 0.0)
                    attenuated_bp = downstream_saturation * attenuation
                    
                    downstream_bp = max(downstream_bp, attenuated_bp)
            
            # Combined backpressure
            combined_bp = max(local_bp, downstream_bp)
            backpressure[stage] = combined_bp
        
        return backpressure
    
    def _measure_stage_saturation(self, family_id: str, stage: Stage) -> float:
        """Measure saturation at stage"""
        db = get_db()
        
        with db.connection() as conn:
            # Queue depth
            cursor = conn.execute("""
                SELECT COUNT(*) FROM task_queue
                WHERE family_id = %s AND current_stage = %s AND status = 'PENDING'
            """, (family_id, stage.value))
            depth = cursor.fetchone()[0]
            
            # Latency p99
            cursor = conn.execute("""
                SELECT PERCENTILE_CONT(0.99) WITHIN GROUP (
                    ORDER BY EXTRACT(EPOCH FROM (NOW() - created_at))
                )
                FROM task_queue
                WHERE family_id = %s AND current_stage = %s AND status = 'PENDING'
                  AND created_at > NOW() - INTERVAL '1 hour'
            """, (family_id, stage.value))
            
            row = cursor.fetchone()
            latency_p99 = float(row[0]) if row and row[0] else 0.0
            
            # Failure rate
            cursor = conn.execute("""
                SELECT 
                    COUNT(CASE WHEN status = 'FAILED' THEN 1 END)::float /
                    NULLIF(COUNT(*)::float, 0)
                FROM task_queue
                WHERE family_id = %s AND current_stage = %s
                  AND updated_at > NOW() - INTERVAL '1 hour'
            """, (family_id, stage.value))
            
            row = cursor.fetchone()
            failure_rate = float(row[0]) if row and row[0] else 0.0
        
        # Compute saturation score
        depth_saturation = min(1.0, depth / 500.0)
        latency_saturation = min(1.0, latency_p99 / 600.0)
        failure_saturation = failure_rate
        
        saturation = max(depth_saturation, latency_saturation, failure_saturation)
        
        return saturation
```

```python
def run_supervisor_control_loop():
    """Main supervisor control loop"""
    
    registry = get_supervisor_registry()
    scaling = get_scaling_governance()
    pressure = get_pressure_control()
    
    cycle = 0
    
    while True:
        try:
            cycle += 1
            
            # Check heartbeats and failover
            registry.check_heartbeats_and_failover()
            
            # Get active families
            db = get_db()
            with db.connection() as conn:
                cursor = conn.execute("""
                    SELECT family_id FROM families WHERE status = 'ACTIVE'
                """)
                families = [row[0] for row in cursor.fetchall()]
            
            # Process each family
            for family_id in families:
                try:
                    # Evaluate pressure mode
                    new_mode = pressure.evaluate_mode_transition(family_id)
                    
                    if new_mode:
                        pressure.transition_mode(
                            family_id,
                            new_mode,
                            "automatic_pressure_adjustment"
                        )
                    
                    # Emit metrics
```

### 9.2 CIRCUIT BREAKER

```python
class CircuitState(str, Enum):
    CLOSED = "CLOSED"
    OPEN = "OPEN"
    HALF_OPEN = "HALF_OPEN"

@dataclass
class CircuitBreaker:
    """Circuit breaker for adapter resilience"""
    failure_threshold: int = 5
    success_threshold: int = 2
    timeout_seconds: int = 60
    
    state: CircuitState = CircuitState.CLOSED
    failure_count: int = 0
    success_count: int = 0
    last_failure_time: Optional[datetime] = None
    opened_at: Optional[datetime] = None
    
    def record_success(self):
        """Record successful call"""
        if self.state == CircuitState.HALF_OPEN:
            self.success_count += 1
            if self.success_count >= self.success_threshold:
                self._close()
        elif self.state == CircuitState.CLOSED:
            self.failure_count = 0
    
    def record_failure(self):
        """Record failed call"""
        self.last_failure_time = datetime.now()
        
        if self.state == CircuitState.HALF_OPEN:
            self._open()
        elif self.state == CircuitState.CLOSED:
            self.failure_count += 1
            if self.failure_count >= self.failure_threshold:
                self._open()
    
    def can_execute(self) -> bool:
        """Check if can execute call"""
        if self.state == CircuitState.CLOSED:
            return True
        elif self.state == CircuitState.OPEN:
            # Check if timeout expired
            if self.opened_at:
                elapsed = (datetime.now() - self.opened_at).total_seconds()
                if elapsed >= self.timeout_seconds:
                    self._half_open()
                    return True
            return False
        elif self.state == CircuitState.HALF_OPEN:
            return True
        return False
    
    def _open(self):
        """Open circuit"""
        self.state = CircuitState.OPEN
        self.opened_at = datetime.now()
        self.failure_count = 0
        self.success_count = 0
        logger.warning("circuit_breaker_opened")
    
    def _half_open(self):
        """Half-open circuit"""
        self.state = CircuitState.HALF_OPEN
        self.success_count = 0
        logger.info("circuit_breaker_half_open")
    
    def _close(self):
        """Close circuit"""
        self.state = CircuitState.CLOSED
        self.failure_count = 0
        self.success_count = 0
        self.opened_at = None
        logger.info("circuit_breaker_closed")
```

```python
@dataclass
class RateLimiter:
    """Token bucket rate limiter"""
    tokens_per_second: float
    max_tokens: int
    
    tokens: float = field(init=False)
    last_update: datetime = field(default_factory=datetime.now)
    
    def __post_init__(self):
        self.tokens = float(self.max_tokens)
    
    def acquire(self, tokens: int = 1) -> bool:
        """Acquire tokens, return True if successful"""
        self._refill()
        
        if self.tokens >= tokens:
            self.tokens -= tokens
            return True
        
        return False
    
    def wait_time(self, tokens: int = 1) -> float:
        """Get wait time in seconds for tokens"""
        self._refill()
        
        if self.tokens >= tokens:
            return 0.0
        
        needed = tokens - self.tokens
        return needed / self.tokens_per_second
    
    def _refill(self):
        """Refill tokens based on elapsed time"""
        now = datetime.now()
        elapsed = (now - self.last_update).total_seconds()
        
        new_tokens = elapsed * self.tokens_per_second
        self.tokens = min(self.max_tokens, self.tokens + new_tokens)
        self.last_update = now
```
